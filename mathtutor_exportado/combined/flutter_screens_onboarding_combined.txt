  Widget build(BuildContext context) {

                    _buildStyleOption(
                      style: VisualStyle.balanced,
                      icon: '⚖️',
                      title: 'Balanceado',

                    const SizedBox(height: 16),

                    _buildStyleOption(
                      style: VisualStyle.minimal,
                      icon: '▫️',

                    SizedBox(width: 12),
                    Expanded(
                      child: Text(
                        'No te preocupes, lo ajustaremos automáticamente mientras usás la app (podés cambiar cuando quieras en Configuración)',
                        style: TextStyle(
                          fontSize: 13,
                          color: Color(0xFF1565C0),
                          height: 1.4,
                        ),
                      ),
                    ),
                  ],
                ),
              ),

              const SizedBox(height: 24),

              // Botón Empezar
              SizedBox(
                width: double.infinity,
                height: 56,
                child: ElevatedButton(
                  onPressed: selectedStyle != null ? _onStart : null,
                  style: ElevatedButton.styleFrom(
                    backgroundColor: const Color(0xFF2196F3),

                    shape: RoundedRectangleBorder(
                      borderRadius: BorderRadius.circular(16),
                    ),
                    elevation: selectedStyle != null ? 4 : 0,
                  ),
                  child: Row(
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: const [
                      Text(
                        'Empezar',
                        style: TextStyle(

                        ),
                      ),
                      SizedBox(width: 8),
                      Icon(Icons.arrow_forward, color: Colors.white, size: 22),

              const SizedBox(height: 8),
            ],
          ),
        ),
      ),

  Widget _buildStyleOption({
    required VisualStyle style,
    required String icon,
    required String title,
    required String description,
    required List<Color> colors,
  }) {
    final isSelected = selectedStyle == style;

    return GestureDetector(
      onTap: () {
        setState(() {
          selectedStyle = style;
        });
      },
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 300),
        curve: Curves.easeOut,
        decoration: BoxDecoration(
          gradient: isSelected
              ? LinearGradient(
                  begin: Alignment.topLeft,
                  end: Alignment.bottomRight,
                  colors: colors,
                )
              : null,
          color: isSelected ? null : Colors.white,
          borderRadius: BorderRadius.circular(20),
          border: Border.all(
            color: isSelected
                ? Colors.transparent
                : const Color(0xFFE0E0E0),
            width: 2,
          ),
          boxShadow: [
            BoxShadow(
              color: isSelected
                  ? colors.first.withOpacity(0.3)
                  : Colors.black.withOpacity(0.05),
              blurRadius: isSelected ? 20 : 10,
              offset: Offset(0, isSelected ? 8 : 4),
            ),
          ],
        ),
        padding: const EdgeInsets.all(24),
        child: Row(
          children: [
            // Ícono
            AnimatedContainer(
              duration: const Duration(milliseconds: 300),
              width: 60,
              height: 60,
              decoration: BoxDecoration(
                color: isSelected
                    ? Colors.white.withOpacity(0.2)
                    : const Color(0xFFF5F7FA),
                borderRadius: BorderRadius.circular(16),
              ),
              child: Center(
                child: Text(
                  icon,
                  style: const TextStyle(fontSize: 32),
                ),
              ),
            ),

            const SizedBox(width: 20),

            // Texto
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    title,
                    style: TextStyle(
                      fontSize: 20,
                      fontWeight: FontWeight.bold,
                      color: isSelected
                          ? Colors.white
                          : const Color(0xFF2C3E50),
                    ),
                  ),
                  const SizedBox(height: 6),
                  Text(
                    description,
                    style: TextStyle(
                      fontSize: 14,
                      color: isSelected
                          ? Colors.white.withOpacity(0.9)
                          : const Color(0xFF7F8C8D),
                      height: 1.3,
                    ),
                  ),
                ],
              ),
            ),

            // Checkmark si está seleccionado
            if (isSelected)
              Container(
                width: 32,
                height: 32,
                decoration: const BoxDecoration(
                  color: Colors.white,
                  shape: BoxShape.circle,
                ),
                child: Icon(
                  Icons.check,
                  color: colors.first,
                  size: 20,
                ),
              ),
          ],
        ),
      ),
    );
  }

  void _onStart() async {
    if (selectedStyle == null) return;

    final provider = context.read<VisualStyleProvider>();
    
    // Marcar onboarding como completado y guardar estilo elegido
    await provider.completeOnboarding(selectedStyle!);

    // Mostrar feedback visual
    if (!mounted) return;

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Row(
          children: [
            const Icon(Icons.check_circle, color: Colors.white),
            const SizedBox(width: 12),
            Text(
              '¡Listo! Configurado como ${provider.styleDisplayName}',
              style: const TextStyle(fontSize: 15),
            ),
          ],
        ),
        backgroundColor: const Color(0xFF4CAF50),
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(12),
        ),
        duration: const Duration(seconds: 2),
      ),
    );

    // Esperar un momento y navegar a home
    await Future.delayed(const Duration(milliseconds: 500));

    if (!mounted) return;

    Navigator.of(context).pushReplacementNamed('/home');
  }
}

// lib/screens/presentation_slide.dart

import 'package:flutter/material.dart';
import 'package:flutter_animate/flutter_animate.dart';
import 'package:provider/provider.dart'; // ✅ NUEVO
import '../services/visual_style_provider.dart'; // ✅ NUEVO

class PresentationSlide extends StatefulWidget {
  final String title;
  final List<String> content;
  final int slideIndex;
  final Widget? customWidget;

  const PresentationSlide({
    super.key,
    required this.title,
    required this.content,
    required this.slideIndex,
    this.customWidget,
  });

  @override
  State<PresentationSlide> createState() => _PresentationSlideState();
}

class _PresentationSlideState extends State<PresentationSlide>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      duration: const Duration(milliseconds: 1200),
      vsync: this,
    );
    _controller.forward();
  }

  @override
  Widget build(BuildContext context) {
    return Center(
      child: SingleChildScrollView(
        padding: const EdgeInsets.all(20),
        child: Container(
          constraints: const BoxConstraints(maxWidth: 700),
          padding: const EdgeInsets.all(30),
          decoration: BoxDecoration(
            gradient: LinearGradient(
              begin: Alignment.topLeft,
              end: Alignment.bottomRight,
              colors: [
                Colors.white,
                Colors.blue.shade50,
              ],
            ),
            borderRadius: BorderRadius.circular(24),
            boxShadow: [
              BoxShadow(
                color: Colors.blue.withOpacity(0.15),
                blurRadius: 30,
                spreadRadius: 5,
                offset: const Offset(0, 10),
              )
            ],
            border: Border.all(
              color: Colors.blue.shade100,
              width: 2,
            ),
          ),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            mainAxisSize: MainAxisSize.min,
            children: [
              Text(
                widget.title,
                style: TextStyle(
                  fontSize: 32,
                  fontWeight: FontWeight.bold,
                  color: Colors.blue.shade700,
                  height: 1.2,
                ),
              )
                  .animate()
                  .fadeIn(duration: 400.ms)
                  .slideX(begin: -0.2, duration: 500.ms, curve: Curves.easeOutCubic),

              const SizedBox(height: 24),

              ...List.generate(widget.content.length, (index) {
                return Padding(
                  padding: const EdgeInsets.only(bottom: 16),
                  child: Row(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      Container(
                        margin: const EdgeInsets.only(top: 6, right: 12),
                        width: 8,
                        height: 8,
                        decoration: BoxDecoration(
                          color: Colors.blue.shade600,
                          shape: BoxShape.circle,
                        ),
                      )
                          .animate()
                          .scale(
                            delay: (300 + index * 150).ms,
                            duration: 300.ms,
                            begin: const Offset(0, 0),
                            curve: Curves.elasticOut,
                          ),
                      Expanded(
                        child: Text(
                          widget.content[index],
                          style: const TextStyle(
                            fontSize: 18,
                            color: Colors.black87,
                            height: 1.6,
                            fontWeight: FontWeight.w400,
                          ),
                        )
                            .animate()
                            .fadeIn(
                              delay: (300 + index * 150).ms,
                              duration: 400.ms,
                            )
                            .slideX(
                              begin: -0.3,
                              delay: (300 + index * 150).ms,
                              duration: 500.ms,
                              curve: Curves.easeOut,
                            ),
                      ),
                    ],
                  ),
                );
              }),

              if (widget.customWidget != null) ...[
                const SizedBox(height: 24),
                widget.customWidget!
                    .animate()
                    .fadeIn(delay: 800.ms, duration: 600.ms)
                    .scale(delay: 800.ms, duration: 500.ms, curve: Curves.easeOut),
              ],
            ],
          ),
        ),
      ),
    );
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }
}

class PresentationScreen extends StatefulWidget {
  const PresentationScreen({super.key});

  @override
  State<PresentationScreen> createState() => _PresentationScreenState();
}

class _PresentationScreenState extends State<PresentationScreen> {
  final PageController _pageController = PageController();
  int _current = 0;

  final List<PresentationSlide> _slides = const [
    PresentationSlide(
      title: '👋 Bienvenida',
      content: [
        'Soy tu tutora de matemáticas.',
        'Resolveremos problemas paso a paso con explicaciones y práctica.',
      ],
      slideIndex: 0,
    ),
    PresentationSlide(
      title: '📚 Aprende con práctica',
      content: [
        'Arrastra bloques para transformar ecuaciones.',
        'Recibirás pistas y evaluación automática.',
      ],
      slideIndex: 1,
    ),
    PresentationSlide(
      title: '🚀 Listo para empezar',
      content: [
        'Comencemos a practicar ahora.',
      ],
      slideIndex: 2,
    ),
  ];

  // ✅ MODIFICADO: Ahora decide si va a onboarding o directo a home
  void _goHome() {
    final styleProvider = context.read<VisualStyleProvider>();
    
    if (styleProvider.hasCompletedOnboarding) {
      // Ya eligió estilo antes, ir directo a home
      print('🏠 Usuario ya completó onboarding, navegando a /home');
      Navigator.of(context).pushReplacementNamed('/home');
    } else {
      // Primera vez, mostrar selección de estilos
      print('🎨 Primera vez, navegando a /style-onboarding');
      Navigator.of(context).pushReplacementNamed('/style-onboarding');
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.grey[50],
      appBar: AppBar(
        title: const Text(
          'Introducción',
          style: TextStyle(color: Colors.white, fontWeight: FontWeight.bold),
        ),
        centerTitle: true,
        flexibleSpace: Container(
          decoration: BoxDecoration(
            gradient: LinearGradient(
              colors: [Colors.blue.shade600, Colors.blue.shade400],
            ),
          ),
        ),
        actions: [
          Padding(
            padding: const EdgeInsets.only(right: 8),
            child: TextButton(
              onPressed: () {
                print('⏭️ Saltando introducción');
                _goHome();
              },
              child: const Text(
                'Saltar',
                style: TextStyle(
                  color: Colors.white,
                  fontWeight: FontWeight.bold,
                  fontSize: 16,
                ),
              ),
            ),
          ),
        ],
      ),
      body: Column(
        children: [
          // Indicador de página
          Padding(
            padding: const EdgeInsets.symmetric(vertical: 16),
            child: Row(
              mainAxisAlignment: MainAxisAlignment.center,
              children: List.generate(
                _slides.length,
                (index) => Container(
                  width: 10,
                  height: 10,
                  margin: const EdgeInsets.symmetric(horizontal: 4),
                  decoration: BoxDecoration(
                    shape: BoxShape.circle,
                    color: _current == index
                        ? Colors.blue.shade600
                        : Colors.grey.shade300,
                  ),
                ),
              ),
            ),
          ),

          // Slides
          Expanded(
            child: PageView.builder(
              controller: _pageController,
              itemCount: _slides.length,
              onPageChanged: (i) {
                print('📄 Cambió a página $i');
                setState(() => _current = i);
              },
              itemBuilder: (context, i) => _slides[i],
            ),
          ),

          // Botones de navegación
          SafeArea(
            child: Padding(
              padding: const EdgeInsets.all(20),
              child: Row(
                children: [
                  // Botón "Atrás" (solo si no estamos en la primera página)
                  if (_current > 0)
                    Expanded(
                      child: OutlinedButton(
                        onPressed: () {
                          _pageController.previousPage(
                            duration: const Duration(milliseconds: 400),
                            curve: Curves.easeOut,
                          );
                        },
                        style: OutlinedButton.styleFrom(
                          padding: const EdgeInsets.symmetric(vertical: 16),
                          side: BorderSide(color: Colors.blue.shade600, width: 2),
                          shape: RoundedRectangleBorder(
                            borderRadius: BorderRadius.circular(12),
                          ),
                        ),
                        child: const Text(
                          '← Atrás',
                          style: TextStyle(fontSize: 16, fontWeight: FontWeight.bold),
                        ),
                      ),
                    ),

                  if (_current > 0) const SizedBox(width: 12),

                  // Botón "Siguiente" o "Comenzar"
                  Expanded(
                    flex: _current == 0 ? 2 : 1,
                    child: ElevatedButton(
                      onPressed: () {
                        if (_current < _slides.length - 1) {
                          _pageController.nextPage(
                            duration: const Duration(milliseconds: 400),
                            curve: Curves.easeOut,
                          );
                        } else {
                          print('🎉 Botón Comenzar presionado');
                          _goHome();
                        }
                      },
                      style: ElevatedButton.styleFrom(
                        backgroundColor: Colors.blue.shade600,
                        foregroundColor: Colors.white,
                        padding: const EdgeInsets.symmetric(vertical: 16),
                        shape: RoundedRectangleBorder(
                          borderRadius: BorderRadius.circular(12),
                        ),
                        elevation: 4,
                      ),
                      child: Text(
                        _current < _slides.length - 1 ? 'Siguiente →' : '🚀 Comenzar',
                        style: const TextStyle(
                          fontSize: 16,
                          fontWeight: FontWeight.bold,
                        ),
                      ),
                    ),
                  ),
                ],
              ),
            ),
          ),
        ],
      ),
    );
  }

  @override
  void dispose() {
    _pageController.dispose();
    super.dispose();
  }
}
