// lib/widgets/draggable_math_piece.dart
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:provider/provider.dart';
import '../models/math_piece.dart';
import '../providers/adaptive_style_provider.dart';

class DraggableMathPiece extends StatefulWidget {
  final MathPiece piece;
  final VoidCallback? onDragStart;
  final Function(Offset)? onDragEnd;
  final VoidCallback? onTap;
  
  const DraggableMathPiece({
    Key? key,
    required this.piece,
    this.onDragStart,
    this.onDragEnd,
    this.onTap,
  }) : super(key: key);

  @override
  State<DraggableMathPiece> createState() => _DraggableMathPieceState();
}

class _DraggableMathPieceState extends State<DraggableMathPiece>
    with SingleTickerProviderStateMixin {
  late AnimationController _pulseController;
  late Animation<double> _pulseAnimation;
  bool _isHovering = false;
  
  @override
  void initState() {
    super.initState();
    _pulseController = AnimationController(
      duration: const Duration(milliseconds: 800),
      vsync: this,
    )..repeat(reverse: true);
    
    _pulseAnimation = Tween<double>(begin: 1.0, end: 1.1).animate(
      CurvedAnimation(parent: _pulseController, curve: Curves.easeInOut),
    );
  }
  
  @override
  void dispose() {
    _pulseController.dispose();
    super.dispose();
  }
  
  @override
  Widget build(BuildContext context) {
    final styleProvider = Provider.of<AdaptiveStyleProvider>(context);
    final config = styleProvider.config;
    
    // Si está bloqueada, no es draggable
    if (widget.piece.isLocked) {
      return _buildPieceWidget(config, isLocked: true);
    }
    
    return GestureDetector(
      onTap: () {
        if (config.hapticFeedback) {
          HapticFeedback.lightImpact();
        }
        widget.onTap?.call();
        styleProvider.trackTap();
      },
      child: Draggable<MathPiece>(
        data: widget.piece,
        feedback: Material(
          color: Colors.transparent,
          child: Transform.scale(
            scale: 1.2,
            child: _buildPieceWidget(config, isDragging: true),
          ),
        ),
        childWhenDragging: Opacity(
          opacity: 0.3,
          child: _buildPieceWidget(config),
        ),
        onDragStarted: () {
          if (config.hapticFeedback) {
            HapticFeedback.mediumImpact();
          }
          widget.onDragStart?.call();
          styleProvider.trackTap();
        },
        onDragEnd: (details) {
          widget.onDragEnd?.call(details.offset);
        },
        child: MouseRegion(
          onEnter: (_) => setState(() => _isHovering = true),
          onExit: (_) => setState(() => _isHovering = false),
          child: _buildPieceWidget(config),
        ),
      ),
    );
  }
  
  Widget _buildPieceWidget(
    StyleConfig config, {
    bool isDragging = false,
    bool isLocked = false,
  }) {
    final shouldPulse = widget.piece.isHighlighted && config.pulseOnHint;
    final shouldGlow = _isHovering && config.glowOnHover;
    
    Widget pieceWidget = AnimatedContainer(
      duration: config.animationDuration,
      width: config.pieceSize,
      height: config.pieceSize,
      padding: EdgeInsets.all(config.padding / 2),
      decoration: BoxDecoration(
        color: config.usePieceGradient ? null : _getPieceColor(config),
        gradient: config.usePieceGradient
            ? LinearGradient(
                begin: Alignment.topLeft,
                end: Alignment.bottomRight,
                colors: [
                  _getPieceColor(config),
                  _getPieceColor(config).withOpacity(0.7),
                ],
              )
            : null,
        borderRadius: BorderRadius.circular(config.pieceBorderRadius),
        border: Border.all(
          color: widget.piece.isHighlighted
              ? Colors.amber
              : Colors.black.withOpacity(0.1),
          width: widget.piece.isHighlighted ? 3 : 2,
        ),
        boxShadow: config.pieceElevation > 0
            ? [
                BoxShadow(
                  color: _getPieceColor(config)
                      .withOpacity(config.pieceShadowOpacity),
                  blurRadius: config.pieceElevation * 2,
                  offset: Offset(0, config.pieceElevation),
                ),
              ]
            : null,
      ),
      child: Center(
        child: Text(
          widget.piece.displayValue,
          style: TextStyle(
            fontSize: config.fontSize,
            fontWeight: config.fontWeight,
            fontFamily: config.fontFamily,
            letterSpacing: config.letterSpacing,
            color: config.usePieceGradient ? Colors.white : Colors.black87,
          ),
        ),
      ),
    );
    
    // Aplicar pulse si está highlighted
    if (shouldPulse) {
      pieceWidget = ScaleTransition(
        scale: _pulseAnimation,
        child: pieceWidget,
      );
    }
    
    // Aplicar glow si está hovering
    if (shouldGlow) {
      pieceWidget = Container(
        decoration: BoxDecoration(
          borderRadius: BorderRadius.circular(config.pieceBorderRadius + 4),
          boxShadow: [
            BoxShadow(
              color: _getPieceColor(config).withOpacity(0.6),
              blurRadius: 20,
              spreadRadius: 4,
            ),
          ],
        ),
        child: pieceWidget,
      );
    }
    
    return pieceWidget;
  }
  
  Color _getPieceColor(StyleConfig config) {
    // Rotar colores según tipo de pieza
    final index = widget.piece.type.index % config.pieceColors.length;
    return config.pieceColors[index];
  }
}

import 'package:flutter/material.dart';

class AvatarWidget extends StatefulWidget {
  final String message;
  final bool isSpeaking;
  final double size;

  const AvatarWidget({
    super.key,
    required this.message,
    required this.isSpeaking,
    this.size = 150,
  });

  @override
  State<AvatarWidget> createState() => _AvatarWidgetState();
}

class _AvatarWidgetState extends State<AvatarWidget>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      duration: const Duration(milliseconds: 500),
      vsync: this,
    );
  }

  @override
  void didUpdateWidget(AvatarWidget oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (widget.isSpeaking && !oldWidget.isSpeaking) {
      _controller.repeat(reverse: true);
    } else if (!widget.isSpeaking && oldWidget.isSpeaking) {
      _controller.stop();
      _controller.value = 0;
    }
  }

  @override
  Widget build(BuildContext context) {
    return Container(
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(20),
        boxShadow: [
          BoxShadow(
            color: Colors.black.withOpacity(0.05),
            blurRadius: 10,
            offset: const Offset(0, 4),
          ),
        ],
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          // Avatar animado
          AnimatedBuilder(
            animation: _controller,
            builder: (context, child) {
              return Transform.scale(
                scale: 1.0 + (_controller.value * 0.1),
                child: Container(
                  width: widget.size,
                  height: widget.size,

                    gradient: LinearGradient(
                      begin: Alignment.topLeft,
                      end: Alignment.bottomRight,

                    border: Border.all(
                      color: Colors.black,

                  child: Stack(
                    children: [
                      // Pelo ondulado arriba
                      Positioned(
                        top: 10,
                        left: widget.size * 0.25,
                        right: widget.size * 0.25,
                        child: CustomPaint(
                          size: Size(widget.size * 0.5, 20),
                          painter: WavyHairPainter(),
                        ),
                      ),
                      
                      // Ojos
                      Positioned(
                        top: widget.size * 0.35,
                        left: widget.size * 0.25,
                        child: _buildEye(),
                      ),
                      Positioned(
                        top: widget.size * 0.35,

class WavyHairPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = const Color(0xFF8B5CF6)
      ..style = PaintingStyle.fill;

    final path = Path();
    path.moveTo(0, size.height);
    
    for (double i = 0; i <= size.width; i += size.width / 3) {
      path.quadraticBezierTo(
        i + size.width / 6,
        0,
        i + size.width / 3,
        size.height,
      );
    }
    
    path.lineTo(size.width, size.height);
    path.close();
    
    canvas.drawPath(path, paint);
    
    // Borde negro
    final borderPaint = Paint()
      ..color = Colors.black
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2;
    
    canvas.drawPath(path, borderPaint);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

import 'package:flutter/material.dart';
import '../utils/visual_styles.dart';

class SmartCube extends StatelessWidget {
  final String content;
  final VisualStyleConfig config;
  final bool isActive;
  final VoidCallback? onTap;

  const SmartCube({
    Key? key,
    required this.content,
    required this.config,
    this.isActive = false,
    this.onTap,
  }) : super(key: key);

  @override
  Widget build(BuildContext context) {
    // Si no es 3D, usamos versión plana
    if (!config.cubeIs3D) {
      return _buildFlatCube();
    }
    
    // Si es 3D, usamos versión con gradiente/sombra
    return _build3DCube();
  }

  Widget _build3DCube() {
    return GestureDetector(
      onTap: onTap,
      child: Container(
        width: config.cubeSize,
        height: config.cubeSize,
        decoration: BoxDecoration(
          gradient: config.buttonHasGradient
              ? LinearGradient(
                  begin: Alignment.topLeft,
                  end: Alignment.bottomRight,
                  colors: [
                    isActive ? config.accentColor : config.primaryColor,
                    isActive 
                      ? config.accentColor.withOpacity(0.6)
                      : config.primaryColor.withOpacity(0.6),
                  ],
                )
              : null,
          color: config.buttonHasGradient 
            ? null 
            : (isActive ? config.accentColor : config.primaryColor),
          borderRadius: BorderRadius.circular(config.cubeBorderRadius),
          boxShadow: [
            BoxShadow(
              color: (isActive ? config.accentColor : config.primaryColor)
                  .withOpacity(0.3),
              blurRadius: 8,
              offset: const Offset(0, 4),
            ),
          ],
        ),
        child: Center(
          child: Text(
            content,
            style: TextStyle(
              color: Colors.white,
              fontSize: config.headingSize,
              fontWeight: FontWeight.bold,
            ),
            textAlign: TextAlign.center,
          ),
        ),
      ),
    );
  }

  Widget _buildFlatCube() {
    return GestureDetector(
      onTap: onTap,
      child: Container(
        width: config.cubeSize,
        height: config.cubeSize,
        decoration: BoxDecoration(
          border: Border.all(
            color: isActive ? config.accentColor : config.secondaryColor,
            width: config.cubeBorderWidth,
          ),
          borderRadius: BorderRadius.circular(config.cubeBorderRadius),

  void _startTyping() async {
    if (_shouldStop || !mounted) return;
    
    setState(() => _isTyping = true);

    // Escribir cada paso
    for (int stepIndex = 0; stepIndex < widget.solution.steps.length; stepIndex++) {
      if (_shouldStop || !mounted) return;
      
      final step = widget.solution.steps[stepIndex];
      
      // Título del paso
      final stepTitle = '\n\nPaso ${step.stepNumber}: ';
      await _typeText(stepTitle, speed: 30);
      if (_shouldStop || !mounted) return;
      
      // Expresión matemática
      await _typeText('${step.expression}\n', speed: 40);
      if (_shouldStop || !mounted) return;
      
      // Explicación
      await _typeText('${step.explanation}', speed: 25);
      if (_shouldStop || !mounted) return;
      
      // Pausa entre pasos
      await Future.delayed(const Duration(milliseconds: 300));
    }

    if (_shouldStop || !mounted) return;

    // Resultado final
    await _typeText('\n\n✅ Solución Final: ', speed: 30);
    if (_shouldStop || !mounted) return;
    
    await _typeText(widget.solution.solution, speed: 50);
    if (_shouldStop || !mounted) return;

      setState(() => _isTyping = false);

  late Animation<double> _scaleAnimation;
  late Animation<double> _fadeAnimation;

  @override
  void initState() {
    super.initState();
    _initAnimations();
    _controller.forward();
  }

  void _initAnimations() {
    final duration = _getAnimationDuration();
    final curve = _getAnimationCurve();
    
    _controller = AnimationController(duration: duration, vsync: this);

    _scaleAnimation = Tween<double>(begin: 0.0, end: 1.0).animate(
      CurvedAnimation(parent: _controller, curve: curve),
    );

    _fadeAnimation = Tween<double>(begin: 0.0, end: 1.0).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeIn),
    );
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  Duration _getAnimationDuration() {
    switch (widget.config.copyTone) {
      case 'fun': return const Duration(milliseconds: 600);
      case 'friendly': return const Duration(milliseconds: 400);
      case 'formal': return const Duration(milliseconds: 200);
      default: return const Duration(milliseconds: 400);
    }
  }

  Curve _getAnimationCurve() {
    switch (widget.config.copyTone) {
      case 'fun': return Curves.elasticOut;
      case 'friendly': return Curves.easeOutCubic;
      case 'formal': return Curves.linear;
      default: return Curves.easeOut;
    }
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _controller,
      builder: (context, child) {
        Widget content = _buildContent();

        if (widget.config.copyTone == 'fun') {
          return Transform.scale(
            scale: _scaleAnimation.value,
            child: Opacity(opacity: _fadeAnimation.value, child: content),
          );
        }
        
        return Opacity(opacity: _fadeAnimation.value, child: content);
      },
    );
  }

  Widget _buildContent() {
    return Container(
      padding: EdgeInsets.all(widget.config.spacing),
      decoration: BoxDecoration(
        gradient: _getGradient(),
        borderRadius: BorderRadius.circular(20),
        border: Border.all(
          color: widget.config.primaryColor.withOpacity(0.3),
          width: 2,
        ),
        boxShadow: widget.config.copyTone == 'fun'
            ? [
                BoxShadow(
                  color: widget.config.primaryColor.withOpacity(0.3),
                  blurRadius: 20,
                  offset: const Offset(0, 8),
                ),
              ]
            : [],
      ),
      child: Text(
        widget.expression,
        style: TextStyle(
          fontSize: _getFontSize(),

      default: return widget.config.titleFontSize * 1.5;

      builder: (context, child) {
        return Transform.translate(
          offset: Offset(_shakeAnimation.value * (1), 0),
          child: child,
        );
      },
      child: GestureDetector(
        onTap: () {
          if (config.hapticFeedback) {
            HapticFeedback.lightImpact();
          }
          widget.onTap?.call();
          styleProvider.trackTap();
        },
        child: Container(
          width: double.infinity,
          padding: EdgeInsets.all(config.padding * 2),
          decoration: BoxDecoration(
            gradient: config.usePieceGradient
                ? LinearGradient(
                    colors: [
                      Colors.blue.shade50,
                      Colors.purple.shade50,
                    ],
                  )
                : null,
            color: config.usePieceGradient ? null : Colors.grey.shade100,
            borderRadius: BorderRadius.circular(config.pieceBorderRadius),
            border: Border.all(
              color: config.pieceColors.first,
              width: 3,
            ),
            boxShadow: config.pieceElevation > 0
                ? [
                    BoxShadow(
                      color: Colors.black.withOpacity(0.1),
                      blurRadius: 10,
                      offset: const Offset(0, 3),
                    ),
                  ]
                : null,
          ),
          child: Wrap(
            alignment: WrapAlignment.center,
            spacing: config.spacing,
            children: parts.map((part) => _buildInteractivePart(part, config)).toList(),
          ),
        ),
      ),
    );
  }
  
  List<String> _parseExpression(String expr) {
    // Separar en números y operadores: "2+2" -> ["2", "+", "2"]
    return expr.split(RegExp(r'(\+|\-|\×|\÷|\=)')).where((s) => s.isNotEmpty).toList();
  }
  
  Widget _buildInteractivePart(String part, StyleConfig config) {
    final isOperator = ['+', '-', '×', '÷', '='].contains(part.trim());
    final isHighlighted = _highlightedPart == part;
    
    return GestureDetector(
      onTap: () {
        if (!isOperator) {
          setState(() => _highlightedPart = part);
          widget.onNumberTap?.call(part);
          
          if (config.hapticFeedback) {
            HapticFeedback.selectionClick();
          }
        }
      },
      child: AnimatedContainer(
        duration: config.animationDuration,
        padding: EdgeInsets.symmetric(
          horizontal: config.padding,
          vertical: config.padding / 2,
        ),
        decoration: BoxDecoration(
          color: isHighlighted
              ? config.pieceColors.first.withOpacity(0.3)
              : Colors.transparent,
          borderRadius: BorderRadius.circular(config.pieceBorderRadius / 2),
          border: isHighlighted
              ? Border.all(color: config.pieceColors.first, width: 2)
              : null,
        ),
        child: Text(
          part,
          style: TextStyle(
            fontSize: config.fontSize * 2.5, // Más grande que piezas normales
            fontWeight: config.fontWeight,
            fontFamily: config.fontFamily,
            letterSpacing: config.letterSpacing * 2,
            color: isOperator ? Colors.grey.shade600 : Colors.black87,
          ),
        ),
      ),
    );
  }
  
  // Método público para shake desde afuera
  void shake() {
    _triggerShake();
  }
}

/// Pintor personalizado para el gráfico
class GraphPainter extends CustomPainter {
  final GraphData graphData;
  final double progress;

  GraphPainter({
    required this.graphData,
    required this.progress,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.blue.shade600
      ..strokeWidth = 2.5
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round;

    final gridPaint = Paint()
      ..color = Colors.grey.shade300
      ..strokeWidth = 1;

    final axisPaint = Paint()
      ..color = Colors.black
      ..strokeWidth = 2;

    // Calcular escala
    final xRange = graphData.xMax - graphData.xMin;
    final yRange = graphData.yMax - graphData.yMin;
    
    final xScale = size.width / xRange;
    final yScale = size.height / yRange;

    // Función para convertir coordenadas del gráfico a canvas
    Offset toCanvas(double x, double y) {
      final canvasX = (x - graphData.xMin) * xScale;
      final canvasY = size.height - (y - graphData.yMin) * yScale;
      return Offset(canvasX, canvasY);
    }

    // Dibujar grid
    final gridInterval = _calculateGridInterval(xRange);
    for (double x = (graphData.xMin / gridInterval).ceil() * gridInterval;
        x <= graphData.xMax;
        x += gridInterval) {
      final point = toCanvas(x, 0);
      canvas.drawLine(
        Offset(point.dx, 0),
        Offset(point.dx, size.height),
        gridPaint,
      );
    }

    for (double y = (graphData.yMin / gridInterval).ceil() * gridInterval;
        y <= graphData.yMax;
        y += gridInterval) {
      final point = toCanvas(0, y);
      canvas.drawLine(
        Offset(0, point.dy),
        Offset(size.width, point.dy),
        gridPaint,
      );
    }

    // Dibujar ejes
    final origin = toCanvas(0, 0);
    
    // Eje X
    canvas.drawLine(
      Offset(0, origin.dy),
      Offset(size.width, origin.dy),
      axisPaint,
    );
    
    // Eje Y
    canvas.drawLine(
      Offset(origin.dx, 0),
      Offset(origin.dx, size.height),
      axisPaint,
    );

    // Dibujar etiquetas de números
    final textPainter = TextPainter(
      textDirection: TextDirection.ltr,
    );

    // Etiquetas eje X
    for (double x = (graphData.xMin / gridInterval).ceil() * gridInterval;
        x <= graphData.xMax;
        x += gridInterval) {
      if (x == 0) continue;
      final point = toCanvas(x, 0);
      textPainter.text = TextSpan(
        text: x.toInt().toString(),
        style: const TextStyle(color: Colors.black, fontSize: 10),
      );
      textPainter.layout();
      textPainter.paint(
        canvas,
        Offset(point.dx - 10, origin.dy + 5),
      );
    }

    // Etiquetas eje Y
    for (double y = (graphData.yMin / gridInterval).ceil() * gridInterval;
        y <= graphData.yMax;
        y += gridInterval) {
      if (y == 0) continue;
      final point = toCanvas(0, y);
      textPainter.text = TextSpan(
        text: y.toInt().toString(),
        style: const TextStyle(color: Colors.black, fontSize: 10),
      );
      textPainter.layout();
      textPainter.paint(
        canvas,
        Offset(origin.dx + 5, point.dy - 10),
      );
    }

    // Dibujar línea de la ecuación (animada)
    if (graphData.xValues.isNotEmpty && graphData.yValues.isNotEmpty) {
      final path = Path();
      final visiblePoints = (graphData.xValues.length * progress).round();
      
      bool firstPoint = true;
      for (int i = 0; i < visiblePoints; i++) {
        final x = graphData.xValues[i];
        final y = graphData.yValues[i];
        
        if (!y.isFinite) continue;
        
        final point = toCanvas(x, y);
        
        if (firstPoint) {
          path.moveTo(point.dx, point.dy);
          firstPoint = false;
        } else {
          path.lineTo(point.dx, point.dy);
        }
      }
      
      canvas.drawPath(path, paint);
    }

    // Dibujar punto de solución
    if (graphData.solutionPoint != null && progress > 0.5) {
      final solutionPaint = Paint()
        ..color = Colors.green.shade700
        ..style = PaintingStyle.fill;

      final solutionPoint = toCanvas(
        graphData.solutionPoint!.x,
        graphData.solutionPoint!.y,
      );

      canvas.drawCircle(solutionPoint, 6, solutionPaint);
      
      // Borde blanco
      final borderPaint = Paint()
        ..color = Colors.white
        ..style = PaintingStyle.stroke
        ..strokeWidth = 2;
      canvas.drawCircle(solutionPoint, 6, borderPaint);
    }
  }

  double _calculateGridInterval(double range) {
    if (range <= 10) return 1;
    if (range <= 20) return 2;
    if (range <= 50) return 5;
    if (range <= 100) return 10;
    return 20;
  }

  @override
  bool shouldRepaint(GraphPainter oldDelegate) {
    return oldDelegate.progress != progress;
  }
}

import 'package:flutter/material.dart';
import '../models/math_piece.dart';

class DropZoneWidget extends StatelessWidget {
  final DropZone dropZone;
  final Function(MathPiece, DropZone) onPieceDropped;
  final Widget child;
  
  const DropZoneWidget({
    Key? key,
    required this.dropZone,
    required this.onPieceDropped,
    required this.child,
  }) : super(key: key);

  @override
  Widget build(BuildContext context) {
    return DragTarget<MathPiece>(
      onWillAccept: (piece) {
        return piece != null && dropZone.canAccept(piece);
      },
      onAccept: (piece) {
        onPieceDropped(piece, dropZone);
      },
      builder: (context, candidateData, rejectedData) {
        final isActive = candidateData.isNotEmpty;
        final isRejected = rejectedData.isNotEmpty;
        
        return AnimatedContainer(
          duration: const Duration(milliseconds: 200),
          decoration: BoxDecoration(
            color: isRejected
                ? Colors.red.withOpacity(0.1)
                : isActive
                    ? Colors.green.withOpacity(0.1)
                    : Colors.grey.withOpacity(0.05),
            borderRadius: BorderRadius.circular(12),
            border: Border.all(
              color: isRejected
                  ? Colors.red
                  : isActive
                      ? Colors.green
                      : Colors.grey.shade300,
              width: isActive || isRejected ? 3 : 2,
            ),
          ),
          child: child,
        );
      },
    );
  }
}
