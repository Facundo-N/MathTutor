#!/usr/bin/env python
"""
Script de prueba para WebSocket de Banny
Envía un mensaje de prueba y verifica la respuesta de OpenAI
"""

import asyncio
import json
import websockets
import sys

async def test_websocket():
    uri = "ws://localhost:8000/ws/chat/test_user_123"
    
    try:
        print(f"🔗 Conectando a {uri}...")
        async with websockets.connect(uri) as websocket:
            print("✅ WebSocket conectado")
            
            # Enviar mensaje de prueba
            test_message = {
                "type": "message",
                "content": "¿Cuál es 2 + 2?",
                "sender": "user"
            }
            
            print(f"📨 Enviando: {test_message['content']}")
            await websocket.send(json.dumps(test_message))
            
            # Recibir respuesta
            print("⏳ Esperando respuesta de OpenAI...")
            response = await asyncio.wait_for(websocket.recv(), timeout=15.0)
            response_data = json.loads(response)
            
            print("\n" + "="*60)
            print("📤 RESPUESTA RECIBIDA:")
            print("="*60)
            print(f"Tipo: {response_data.get('type')}")
            print(f"Contenido: {response_data.get('content')}")
            print(f"Remitente: {response_data.get('sender')}")
            print(f"Modelo usado: {response_data.get('model_used')}")
            print(f"Fuente: {response_data.get('source')}")
            print("="*60)
            
            if response_data.get('source') == 'openai':
                print("✅ ¡OpenAI está funcionando correctamente!")
            else:
                print("⚠️ La respuesta no viene de OpenAI")
                
    except asyncio.TimeoutError:
        print("❌ Timeout esperando respuesta (>15s)")
        sys.exit(1)
    except Exception as e:
        print(f"❌ Error: {e}")
        sys.exit(1)

if __name__ == "__main__":
    asyncio.run(test_websocket())
