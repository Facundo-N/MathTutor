// Validador inteligente de respuestas matemáticas

class AnswerValidationResult {
  final bool isCorrect;
  final String normalizedAnswer;
  final String? expectedAnswer;
  final String? feedback;

  AnswerValidationResult.correct(
    this.normalizedAnswer, {
    this.feedback,
  })  : isCorrect = true,
        expectedAnswer = null;

  AnswerValidationResult.incorrect(
    this.normalizedAnswer,
    this.expectedAnswer, {
    this.feedback,
  }) : isCorrect = false;

  AnswerValidationResult.invalid(
    this.normalizedAnswer, {
    this.feedback,
  })  : isCorrect = false,
        expectedAnswer = null;
}

class AnswerValidator {
  /// Valida respuesta del usuario contra la respuesta esperada
  static AnswerValidationResult validate(String userInput, dynamic expectedAnswer) {
    // Limpiar input básico
    String cleaned = userInput.trim().toLowerCase()
        .replaceAll(' ', '')
        .replaceAll(',', '.');

    if (cleaned.isEmpty) {
      return AnswerValidationResult.invalid(
        '',
        feedback: 'La respuesta está vacía',
      );
    }

    // Intentar diferentes formatos

    // 1. NÚMERO ENTERO
    final intResult = _tryParseInt(cleaned, expectedAnswer);
    if (intResult != null) return intResult;

    // 2. NÚMERO DECIMAL
    final doubleResult = _tryParseDouble(cleaned, expectedAnswer);
    if (doubleResult != null) return doubleResult;

    // 3. FORMATO "x = valor"
    final equationResult = _tryParseEquation(cleaned, expectedAnswer);
    if (equationResult != null) return equationResult;

    // 4. FRACCIÓN "a/b"
    final fractionResult = _tryParseFraction(cleaned, expectedAnswer);
    if (fractionResult != null) return fractionResult;

    // 5. PORCENTAJE "50%"
    final percentResult = _tryParsePercent(cleaned, expectedAnswer);
    if (percentResult != null) return percentResult;

    // 6. RESPUESTA TEXTUAL (sí/no, verdadero/falso)
    final textResult = _tryParseText(cleaned, expectedAnswer);
    if (textResult != null) return textResult;

    // No se pudo parsear
    return AnswerValidationResult.incorrect(
      cleaned,
      expectedAnswer.toString(),
      feedback: 'No reconozco ese formato. Probá con un número.',
    );
  }

  // ============================================
  // PARSEADORES ESPECÍFICOS
  // ============================================

  /// Intentar parsear como entero
  static AnswerValidationResult? _tryParseInt(String input, dynamic expected) {
    final value = int.tryParse(input);
    if (value == null) return null;

    // Comparar con diferentes tipos
    if (expected is int && value == expected) {
      return AnswerValidationResult.correct(value.toString());
    } else if (expected is double && value.toDouble() == expected) {
      return AnswerValidationResult.correct(value.toString());
    } else if (expected is String && value.toString() == expected) {
      return AnswerValidationResult.correct(value.toString());
    }

    return AnswerValidationResult.incorrect(
      value.toString(),
      expected.toString(),
    );
  }

  /// Intentar parsear como decimal
  static AnswerValidationResult? _tryParseDouble(String input, dynamic expected) {
    final value = double.tryParse(input);
    if (value == null) return null;

    const epsilon = 0.01; // Tolerancia para decimales

    // Comparar según tipo de expected
    if (expected is int) {
      final diff = (value - expected.toDouble()).abs();
      if (diff < epsilon) {
        return AnswerValidationResult.correct(
          value.toStringAsFixed(2).replaceAll(RegExp(r'\.?0+$'), ''),
        );
      }
    } else if (expected is double) {
      final diff = (value - expected).abs();
      if (diff < epsilon) {
        return AnswerValidationResult.correct(
          value.toStringAsFixed(2).replaceAll(RegExp(r'\.?0+$'), ''),
        );
      }
    } else if (expected is String) {
      final expectedNum = double.tryParse(expected);
      if (expectedNum != null && (value - expectedNum).abs() < epsilon) {
        return AnswerValidationResult.correct(value.toString());
      }
    }

    return AnswerValidationResult.incorrect(
      value.toString(),
      expected.toString(),
    );
  }

  /// Intentar parsear ecuación "x=2" o "x = 2.5"
  static AnswerValidationResult? _tryParseEquation(String input, dynamic expected) {
    // Regex: x = número
    final regex = RegExp(r'^x=([\d.+-]+)$');
    final match = regex.firstMatch(input);

    if (match == null) return null;

    final valueStr = match.group(1);
    if (valueStr == null) return null;

    final value = double.tryParse(valueStr);
    if (value == null) return null;

    const epsilon = 0.01;

    // Comparar según tipo
    if (expected is int) {
      if ((value - expected.toDouble()).abs() < epsilon) {
        return AnswerValidationResult.correct('x = $value');
      }
    } else if (expected is double) {
      if ((value - expected).abs() < epsilon) {
        return AnswerValidationResult.correct('x = $value');
      }
    }

    return AnswerValidationResult.incorrect(
      'x = $value',
      'x = $expected',
    );
  }

  /// Intentar parsear fracción "3/4"
  static AnswerValidationResult? _tryParseFraction(String input, dynamic expected) {
    final regex = RegExp(r'^(\d+)/(\d+)$');
    final match = regex.firstMatch(input);

    if (match == null) return null;

    final numStr = match.group(1);
    final denStr = match.group(2);

    if (numStr == null || denStr == null) return null;

    final numerator = int.tryParse(numStr);
    final denominator = int.tryParse(denStr);

    if (numerator == null || denominator == null || denominator == 0) return null;

    final decimal = numerator / denominator;
    const epsilon = 0.01;

    // Comparar según tipo
    if (expected is int) {
      if ((decimal - expected.toDouble()).abs() < epsilon) {
        return AnswerValidationResult.correct(
          '$numerator/$denominator',
          feedback: 'Correcto! ($numerator/$denominator = ${decimal.toStringAsFixed(2)})',
        );
      }
    } else if (expected is double) {
      if ((decimal - expected).abs() < epsilon) {
        return AnswerValidationResult.correct(
          '$numerator/$denominator',
          feedback: 'Correcto! ($numerator/$denominator = ${decimal.toStringAsFixed(2)})',
        );
      }
    } else if (expected is String) {
      // Comparar como fracciones
      if (expected == input) {
        return AnswerValidationResult.correct('$numerator/$denominator');
      }
    }

    return AnswerValidationResult.incorrect(
      '$numerator/$denominator',
      expected.toString(),
    );
  }

  /// Intentar parsear porcentaje "50%"
  static AnswerValidationResult? _tryParsePercent(String input, dynamic expected) {
    final regex = RegExp(r'^(\d+(?:\.\d+)?)%$');
    final match = regex.firstMatch(input);

    if (match == null) return null;

    final valueStr = match.group(1);
    if (valueStr == null) return null;

    final percent = double.tryParse(valueStr);
    if (percent == null) return null;

    final decimal = percent / 100;
    const epsilon = 0.01;

    // Comparar según tipo
    if (expected is int || expected is double) {
      final expectedDouble = expected is int ? expected.toDouble() : expected as double;

      // Si esperamos decimal (0.5) y escriben "50%"
      if ((decimal - expectedDouble).abs() < epsilon) {
        return AnswerValidationResult.correct(
          '$percent%',
          feedback: 'Correcto! $percent% = $decimal',
        );
      }
      // Si esperamos porcentaje (50) y escriben "50%"
      if ((percent - expectedDouble).abs() < epsilon) {
        return AnswerValidationResult.correct('$percent%');
      }
    }

    return AnswerValidationResult.incorrect(
      '$percent%',
      expected.toString(),
    );
  }

  /// Intentar parsear texto (sí/no, verdadero/falso)
  static AnswerValidationResult? _tryParseText(String input, dynamic expected) {
    final positiveWords = ['si', 'sí', 'yes', 'verdadero', 'true', 'correcto'];
    final negativeWords = ['no', 'falso', 'false', 'incorrecto'];

    if (!positiveWords.contains(input) && !negativeWords.contains(input)) {
      return null;
    }

    final userSaysYes = positiveWords.contains(input);

    if (expected is bool) {
      if (userSaysYes == expected) {
        return AnswerValidationResult.correct(userSaysYes ? 'Sí' : 'No');
      }
      return AnswerValidationResult.incorrect(
        userSaysYes ? 'Sí' : 'No',
        expected ? 'Sí' : 'No',
      );
    }

    return null;
  }

  // ============================================
  // UTILIDADES
  // ============================================

  /// Normalizar respuesta para mostrar
  static String normalizeForDisplay(dynamic answer) {
    if (answer is int) return answer.toString();
    if (answer is double) {
      // Eliminar .0 si es entero
      if (answer == answer.toInt()) {
        return answer.toInt().toString();
      }
      return answer.toStringAsFixed(2).replaceAll(RegExp(r'\.?0+$'), '');
    }
    return answer.toString();
  }

  /// Detectar si la respuesta del usuario está "cerca"
  static bool isClose(String userInput, dynamic expectedAnswer, {double tolerance = 0.1}) {
    final userNum = double.tryParse(userInput.replaceAll(',', '.'));

    if (userNum == null) return false;

    if (expectedAnswer is int || expectedAnswer is double) {
      final expectedDouble = expectedAnswer is int 
          ? expectedAnswer.toDouble() 
          : expectedAnswer as double;
      final diff = (userNum - expectedDouble).abs();
      final maxDiff = expectedDouble.abs() * tolerance;
      return diff <= maxDiff;
    }

    return false;
  }
}

// lib/services/hyperpersonalized_tutor_service.dart

import 'package:web_socket_channel/web_socket_channel.dart';
import 'dart:convert';
import 'dart:async';
import '../models/tutor_action.dart';

class HyperpersonalizedTutorService {
  // ✅ Configurado para conectarse desde emulador/dispositivo
  static const String wsUrl = 'ws://192.168.0.11:8000/ws/tutor';
  static const Duration connectionTimeout = Duration(seconds: 10);
  
  WebSocketChannel? _channel;
  final String userId;
  StreamSubscription? _streamSubscription;
  
  // Callback cuando llega una TutorAction
  final void Function(TutorAction action) onTutorAction;
  
  // Stream controller para estados
  final _connectionController = StreamController<bool>.broadcast();
  Stream<bool> get connectionStream => _connectionController.stream;
  
  bool _isConnected = false;
  bool _isConnecting = false;
  bool get isConnected => _isConnected;
  
  HyperpersonalizedTutorService({
    required this.userId,
    required this.onTutorAction,
  });
  
  /// Conectar al WebSocket con timeout y retry
  Future<void> connect() async {
    // Evitar múltiples intentos simultáneos
    if (_isConnecting || _isConnected) {
      print('⚠️ Ya está conectando o conectado');
      return;
    }
    
    _isConnecting = true;
    
    try {
      print('🔌 Conectando WebSocket: $wsUrl/$userId');
      
      _channel = WebSocketChannel.connect(
        Uri.parse('$wsUrl/$userId'),
      );
      
      // Esperar a que se establezca la conexión con timeout
      await _channel!.ready.timeout(
        connectionTimeout,
        onTimeout: () {
          throw Exception('WebSocket connection timeout después de ${connectionTimeout.inSeconds}s');
        },
      );
      
      _isConnected = true;
      _isConnecting = false;
      _connectionController.add(true);
      
      print('✅ WebSocket conectado correctamente');
      
      // Escuchar mensajes del backend
      _streamSubscription = _channel!.stream.listen(
        _handleMessage,
        onError: (error, stackTrace) {
          print('❌ WebSocket stream error: $error');
          print('Stack: $stackTrace');
          _handleDisconnect();
        },
        onDone: () {
          print('⚠️ WebSocket cerrado por el servidor');
          _handleDisconnect();
        },
        cancelOnError: false,
      );
      
    } on TimeoutException catch (e) {
      print('⏱️ Timeout conectando WebSocket: $e');
      _handleDisconnect();
    } catch (e) {
      print('❌ Error conectando WebSocket: $e');
      _handleDisconnect();
      // No relanzar, simplemente logged
    } finally {
      _isConnecting = false;
    }
  }
  
  /// Manejar desconexión
  void _handleDisconnect() {
    _isConnected = false;
    _connectionController.add(false);
    try {
      _streamSubscription?.cancel();
      _channel?.sink.close();
    } catch (e) {
      print('⚠️ Error cerrando canal: $e');
    }
  }
  
  /// Manejar mensajes del backend
  void _handleMessage(dynamic message) {
    try {
      print('📩 Mensaje recibido: $message');
      
      final data = json.decode(message);
      final action = TutorAction.fromJson(data);
      
      print('🎯 TutorAction: ${action.type} - "${action.speech}"');
      
      // Ejecutar callback de forma segura
      try {
        onTutorAction(action);
      } catch (e) {
        print('❌ Error en callback onTutorAction: $e');
      }
      
    } catch (e) {
      print('❌ Error procesando mensaje WebSocket: $e');
    }
  }
  
  /// Enviar señal al backend (opcional)
  void sendSignal(Map<String, dynamic> signal) {
    try {
      if (_isConnected && _channel != null) {
        _channel!.sink.add(json.encode(signal));
        print('📤 Señal enviada: ${signal['type']}');
      } else {
        print('⚠️ WebSocket no conectado, no se puede enviar señal');
      }
    } catch (e) {
      print('❌ Error enviando señal: $e');
    }
  }
  
  /// Desconectar
  void disconnect() {
    _handleDisconnect();
    print('👋 WebSocket desconectado manualmente');
  }
  
  void dispose() {
    disconnect();
    _connectionController.close();
  }
}

import 'package:http/http.dart' as http;
import 'dart:convert';

/// Modelos para la API
class MathStep {
  final int stepNumber;
  final String expression;
  final String explanation;
  final String? latex;

  MathStep({
    required this.stepNumber,
    required this.expression,
    required this.explanation,
    this.latex,
  });

  factory MathStep.fromJson(Map<String, dynamic> json) {
    return MathStep(
      stepNumber: json['step_number'],
      expression: json['expression'],
      explanation: json['explanation'],
      latex: json['latex'],
    );
  }
}

// ✅ Modelo para punto en el gráfico
class GraphPoint {
  final double x;
  final double y;

  GraphPoint({required this.x, required this.y});

  factory GraphPoint.fromJson(Map<String, dynamic> json) {
    return GraphPoint(
      x: (json['x'] ?? 0).toDouble(),
      y: (json['y'] ?? 0).toDouble(),
    );
  }
}

// ✅ Modelo para datos del gráfico
class GraphData {
  final bool canGraph;
  final List<double> xValues;
  final List<double> yValues;
  final GraphPoint? solutionPoint;
  final String equation;
  final double xMin;
  final double xMax;
  final double yMin;
  final double yMax;
  final String? error;

  GraphData({
    required this.canGraph,
    this.xValues = const [],
    this.yValues = const [],
    this.solutionPoint,
    this.equation = '',
    this.xMin = -10,
    this.xMax = 10,
    this.yMin = -10,
    this.yMax = 10,
    this.error,
  });

  factory GraphData.fromJson(Map<String, dynamic> json) {
    print('🔧 Parseando GraphData...');
    print('   can_graph: ${json['can_graph']}');
    print('   x_values length: ${json['x_values']?.length ?? 0}');
    print('   y_values length: ${json['y_values']?.length ?? 0}');
    
    return GraphData(
      canGraph: json['can_graph'] ?? false,
      xValues: json['x_values'] != null
          ? List<double>.from(json['x_values'].map((x) => x.toDouble()))
          : [],
      yValues: json['y_values'] != null
          ? List<double>.from(json['y_values'].map((y) => y.toDouble()))
          : [],
      solutionPoint: json['solution_point'] != null
          ? GraphPoint.fromJson(json['solution_point'])
          : null,
      equation: json['equation'] ?? '',
      xMin: (json['x_min'] ?? -10).toDouble(),
      xMax: (json['x_max'] ?? 10).toDouble(),
      yMin: (json['y_min'] ?? -10).toDouble(),
      yMax: (json['y_max'] ?? 10).toDouble(),
      error: json['error'],
    );
  }
}

/// Servicio para comunicarse con el backend Python
class MathEngineService {
  // URL del servidor
  // Para emulador Android: http://10.0.2.2:8000
  // Para dispositivo físico: http://TU_IP_LOCAL:8000
  static const String baseUrl = 'http://10.0.2.2:8000';

  static final http.Client _client = http.Client();

  /// Resuelve una expresión matemática con pasos
  static Future<MathSolution> solveMath({
    required String expression,
    bool showSteps = true,
  }) async {
    try {
      print('📤 Enviando al backend: $expression');

      final response = await _client.post(
        Uri.parse('$baseUrl/api/solve'),
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json',
        },
        body: jsonEncode({
          'expression': expression,
          'showsteps': showSteps, // ✅ CORREGIDO: sin guión bajo (como espera el backend)
        }),
      ).timeout(
        const Duration(seconds: 10),
        onTimeout: () => throw Exception('Timeout - El servidor no respondió'),
      );

      print('📥 Response status: ${response.statusCode}');
      print('📥 Response body (primeros 500 chars): ${response.body.substring(0, response.body.length > 500 ? 500 : response.body.length)}');

      if (response.statusCode == 200) {
        final jsonData = jsonDecode(response.body);
        final solution = MathSolution.fromJson(jsonData);
        
        print('✅ Solución parseada: ${solution.solution}');
        print('✅ GraphData disponible: ${solution.graphData != null}');
        
        if (solution.graphData != null) {
          print('📊 Gráfico canGraph: ${solution.graphData!.canGraph}');
          if (solution.graphData!.canGraph) {
            print('📊 Ecuación: ${solution.graphData!.equation}');
            print('📊 Puntos: ${solution.graphData!.xValues.length}');
          } else if (solution.graphData!.error != null) {
            print('❌ Error en gráfico: ${solution.graphData!.error}');
          }
        }
        
        return solution;
      } else {
        throw Exception('Error ${response.statusCode}: ${response.body}');
      }
    } catch (e) {
      print('❌ Error en solveMath: $e');
      rethrow;
    }
  }

  /// Obtiene pistas progresivas
  static Future<List<String>> getHints({
    required String expression,
    int level = 0,
  }) async {
    try {
      final response = await _client.post(
        Uri.parse('$baseUrl/api/hints'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'expression': expression,
          'level': level,
        }),
      );

      if (response.statusCode == 200) {
        final jsonData = jsonDecode(response.body);
        return List<String>.from(jsonData['hints']);
      } else {
        throw Exception('Error al obtener pistas');
      }
    } catch (e) {
      print('❌ Error en getHints: $e');
      rethrow;
    }
  }

  /// Verifica conexión con el servidor
  static Future<bool> checkConnection() async {
    try {
      final response = await _client.get(
        Uri.parse('$baseUrl/health'),
      ).timeout(const Duration(seconds: 5));

      return response.statusCode == 200;
    } catch (e) {
      print('❌ Sin conexión con el servidor: $e');
      return false;
    }
  }
}

// Parser matemático mejorado para MathTutor

import 'dart:math' as math;

class MathProblem {
  final String expression;
  final String operation;
  final dynamic num1;
  final dynamic num2;
  final dynamic answer;
  final String visualType;
  final List<String> steps;
  final String difficulty; // 'primaria', 'secundaria'

  MathProblem({
    required this.expression,
    required this.operation,
    required this.num1,
    required this.num2,
    required this.answer,
    required this.visualType,
    required this.steps,
    required this.difficulty,
  });

  Map<String, dynamic> toJson() => {
        'expression': expression,
        'operation': operation,
        'num1': num1,
        'num2': num2,
        'answer': answer,
        'visualType': visualType,
        'steps': steps,
        'difficulty': difficulty,
      };
}

class AdvancedMathParser {
  // ============================================
  // PARSER PRINCIPAL
  // ============================================

  static MathProblem? parse(String text) {
    text = text.replaceAll(' ', '').replaceAll(',', '.').trim();
    // ignore: avoid_print
    print('📝 Parseando: $text');

    // Intentar operaciones básicas primero
    MathProblem? result;

    result = _parseSuma(text);
    if (result != null) return result;

    result = _parseResta(text);
    if (result != null) return result;

    result = _parseMultiplicacion(text);
    if (result != null) return result;

    result = _parseDivision(text);
    if (result != null) return result;

    result = _parsePotencia(text);
    if (result != null) return result;

    result = _parseRaiz(text);
    if (result != null) return result;

    result = _parseEcuacionLineal(text);
    if (result != null) return result;

    // ignore: avoid_print
    print('❌ No se detectó operación matemática');
    return null;
  }

  // ============================================
  // OPERACIONES BÁSICAS (PRIMARIA)
  // ============================================

  static MathProblem? _parseSuma(String text) {
    final regex = RegExp(r'^(\d+)\+(\d+)$');
    final match = regex.firstMatch(text);

    if (match != null) {
      // ✅ NULL SAFETY: Verificar antes de usar
      final num1Str = match.group(1);
      final num2Str = match.group(2);

      if (num1Str == null || num2Str == null) return null;

      final num1 = int.parse(num1Str);
      final num2 = int.parse(num2Str);
      final answer = num1 + num2;

      return MathProblem(
        expression: '$num1+$num2',
        operation: '+',
        num1: num1,
        num2: num2,
        answer: answer,
        visualType: 'sum',
        steps: [
          'Sumamos $num1 + $num2',
          'Resultado: $answer'
        ],
        difficulty: 'primaria',
      );
    }
    return null;
  }

  static MathProblem? _parseResta(String text) {
    final regex = RegExp(r'^(\d+)-(\d+)$');
    final match = regex.firstMatch(text);

    if (match != null) {
      // ✅ NULL SAFETY: Verificar antes de usar
      final num1Str = match.group(1);
      final num2Str = match.group(2);

      if (num1Str == null || num2Str == null) return null;

      final num1 = int.parse(num1Str);
      final num2 = int.parse(num2Str);
      final answer = num1 - num2;

      return MathProblem(
        expression: '$num1-$num2',
        operation: '-',
        num1: num1,
        num2: num2,
        answer: answer,
        visualType: 'subtract',
        steps: [
          'Restamos $num1 - $num2',
          'Resultado: $answer'
        ],
        difficulty: 'primaria',
      );
    }
    return null;
  }

  static MathProblem? _parseMultiplicacion(String text) {
    // Soporta: 7*8, 7x8, 7×8
    final regex = RegExp(r'^(\d+)[*x×](\d+)$');
    final match = regex.firstMatch(text);

    if (match != null) {
      // ✅ NULL SAFETY: Verificar antes de usar
      final num1Str = match.group(1);
      final num2Str = match.group(2);

      if (num1Str == null || num2Str == null) return null;

      final num1 = int.parse(num1Str);
      final num2 = int.parse(num2Str);
      final answer = num1 * num2;

      return MathProblem(
        expression: '$num1×$num2',
        operation: '×',
        num1: num1,
        num2: num2,
        answer: answer,
        visualType: 'multiply',
        steps: [
          'Multiplicamos $num1 × $num2',
          'Esto es lo mismo que sumar $num1, $num2 veces',
          'Resultado: $answer'
        ],
        difficulty: 'primaria',
      );
    }
    return null;
  }

  static MathProblem? _parseDivision(String text) {
    // Soporta: 20/4, 20÷4
    final regex = RegExp(r'^(\d+)[/÷](\d+)$');
    final match = regex.firstMatch(text);

    if (match != null) {
      // ✅ NULL SAFETY: Verificar antes de usar
      final num1Str = match.group(1);
      final num2Str = match.group(2);

      if (num1Str == null || num2Str == null) return null;

      final num1 = int.parse(num1Str);
      final num2 = int.parse(num2Str);

      if (num2 == 0) {
        return MathProblem(
          expression: '$num1÷0',
          operation: '÷',
          num1: num1,
          num2: 0,
          answer: 'indefinido',
          visualType: 'error',
          steps: ['No se puede dividir por 0'],
          difficulty: 'primaria',
        );
      }

      final cociente = num1 ~/ num2;
      final resto = num1 % num2;

      final answer = resto == 0 ? cociente.toString() : '$cociente (resto $resto)';

      return MathProblem(
        expression: '$num1÷$num2',
        operation: '÷',
        num1: num1,
        num2: num2,
        answer: answer,
        visualType: 'divide',
        steps: [
          'Dividimos $num1 ÷ $num2',
          '¿Cuántas veces entra $num2 en $num1?',
          'Cociente: $cociente',
          if (resto > 0) 'Resto: $resto'
        ],
        difficulty: 'primaria',
      );
    }
    return null;
  }

  // ============================================
  // OPERACIONES INTERMEDIAS (SECUNDARIA)
  // ============================================

  static MathProblem? _parsePotencia(String text) {
    // Soporta: 2^3, 2**3
    final regex = RegExp(r'^(\d+)[\^*]+\*?(\d+)$');
    final match = regex.firstMatch(text);

    if (match != null) {
      // ✅ NULL SAFETY: Verificar antes de usar
      final baseStr = match.group(1);
      final exponenteStr = match.group(2);

      if (baseStr == null || exponenteStr == null) return null;

      final base = int.parse(baseStr);
      final exponente = int.parse(exponenteStr);
      final answer = math.pow(base, exponente);

      return MathProblem(
        expression: '$base^$exponente',
        operation: '^',
        num1: base,
        num2: exponente,
        answer: answer,
        visualType: 'power',
        steps: [
          'Calculamos $base^$exponente',
          'Esto es: $base × $base × ... ($exponente veces)',
          'Resultado: $answer'
        ],
        difficulty: 'secundaria',
      );
    }
    return null;
  }

  static MathProblem? _parseRaiz(String text) {
    // Soporta: √16, sqrt(16)
    final regex = RegExp(r'^[√]?(sqrt)?\(?(\d+)\)?$');
    final match = regex.firstMatch(text);

    if (match != null && (text.contains('√') || text.contains('sqrt'))) {
      // ✅ NULL SAFETY: Verificar antes de usar
      final numStr = match.group(2);

      if (numStr == null) return null;

      final num = int.parse(numStr);
      final answer = math.sqrt(num);
      final esExacta = answer == answer.toInt();

      return MathProblem(
        expression: '√$num',
        operation: '√',
        num1: num,
        num2: 0,
        answer: esExacta ? answer.toInt() : answer.toStringAsFixed(2),
        visualType: 'sqrt',
        steps: [
          'Calculamos √$num',
          '¿Qué número multiplicado por sí mismo da $num?',
          'Resultado: ${esExacta ? answer.toInt() : answer.toStringAsFixed(2)}'
        ],
        difficulty: 'secundaria',
      );
    }
    return null;
  }

  static MathProblem? _parseEcuacionLineal(String text) {
    // Soporta: 2x+3=7, 5x-2=13
    final regex = RegExp(r'^(\d+)x([+-])(\d+)=(\d+)$');
    final match = regex.firstMatch(text);

    if (match != null) {
      // ✅ NULL SAFETY: Verificar antes de usar
      final aStr = match.group(1);
      final operador = match.group(2);
      final bStr = match.group(3);
      final cStr = match.group(4);

      if (aStr == null || operador == null || bStr == null || cStr == null) {
        return null;
      }

      final a = int.parse(aStr);
      final b = int.parse(bStr);
      final c = int.parse(cStr);

      final bConSigno = operador == '+' ? b : -b;
      final x = (c - bConSigno) / a;

      return MathProblem(
        expression: '${a}x$operador$b=$c',
        operation: 'ecuacion_lineal',
        num1: a,
        num2: bConSigno,
        answer: x.toStringAsFixed(2),
        visualType: 'equation',
        steps: [
          'Ecuación: ${a}x$operador$b=$c',
          'Paso 1: Despejamos x',
          'Restamos $bConSigno de ambos lados: ${a}x = ${c - bConSigno}',
          'Dividimos por $a: x = ${(c - bConSigno) / a}',
          'Solución: x = ${x.toStringAsFixed(2)}'
        ],
        difficulty: 'secundaria',
      );
    }
    return null;
  }

  // ============================================
  // EVALUADOR DE EXPRESIONES CON PARÉNTESIS
  // ============================================

  static double? evaluarExpresion(String expr) {
    try {
      expr = expr.replaceAll(' ', '');
      // Implementación simplificada - para casos complejos usar un parser real
      // Por ahora evalúa expresiones básicas sin paréntesis

      // Orden de operaciones: *, / antes que +, -
      final tokens = _tokenize(expr);
      return _evaluate(tokens);
    } catch (e) {
      // ignore: avoid_print
      print('Error al evaluar: $e');
      return null;
    }
  }

  static List<String> _tokenize(String expr) {
    final tokens = <String>[];
    var currentNum = '';

    for (var i = 0; i < expr.length; i++) {
      final char = expr[i];
      if (RegExp(r'[0-9.]').hasMatch(char)) {
        currentNum += char;
      } else if (RegExp(r'[+\-*/^]').hasMatch(char)) {
        if (currentNum.isNotEmpty) {
          tokens.add(currentNum);
          currentNum = '';
        }
        tokens.add(char);
      }
    }

    if (currentNum.isNotEmpty) {
      tokens.add(currentNum);
    }

    return tokens;
  }

  static double? _evaluate(List<String> tokens) {
    if (tokens.isEmpty) return null;

    // Procesar multiplicación y división primero
    var i = 0;
    while (i < tokens.length) {
      if (i > 0 && i < tokens.length - 1) {
        if (tokens[i] == '*' || tokens[i] == '/') {
          final left = double.parse(tokens[i - 1]);
          final right = double.parse(tokens[i + 1]);
          final result = tokens[i] == '*' ? left * right : left / right;

          tokens.removeRange(i - 1, i + 2);
          tokens.insert(i - 1, result.toString());
          i = 0;
          continue;
        }
      }
      i++;
    }

    // Procesar suma y resta
    i = 0;
    while (i < tokens.length) {
      if (i > 0 && i < tokens.length - 1) {
        if (tokens[i] == '+' || tokens[i] == '-') {
          final left = double.parse(tokens[i - 1]);
          final right = double.parse(tokens[i + 1]);
          final result = tokens[i] == '+' ? left + right : left - right;

          tokens.removeRange(i - 1, i + 2);
          tokens.insert(i - 1, result.toString());
          i = 0;
          continue;
        }
      }
      i++;
    }

    return tokens.length == 1 ? double.parse(tokens[0]) : null;
  }

  // ============================================
  // UTILIDADES
  // ============================================

  static String getDifficultyLevel(String expression) {
    if (expression.contains('x') || expression.contains('^')) {
      return 'secundaria';
    }
    return 'primaria';
  }

  static List<String> getHintsForOperation(String operation) {
    switch (operation) {
      case '+':
        return ['Contá hacia adelante', 'Podés usar tus dedos'];
      case '-':
        return ['Contá hacia atrás', 'Pensá en quitar objetos'];
      case '×':
        return ['Usá la tabla de multiplicar', 'Es una suma repetida'];
      case '÷':
        return ['¿Cuántas veces entra?', 'Usá la multiplicación al revés'];
      case '^':
        return ['Multiplicá el número por sí mismo', 'Ej: 2³ = 2×2×2'];
      case '√':
        return ['¿Qué número al cuadrado da este?', 'Probá algunos números'];
      default:
        return ['Seguí los pasos', 'Andá de a poco'];
    }
  }
}

import 'dart:convert';
import 'dart:async';
import 'package:flutter/foundation.dart';
import 'package:http/http.dart' as http;
import 'package:uuid/uuid.dart';
import '../models/user_signal.dart';

class SignalCollector {
  // Backend URL
  static const String baseUrl = 'http://192.168.0.11:8000';
  static const int bufferMaxSize = 50;
  static const Duration flushInterval = Duration(seconds: 5);
  
  // ⭐ NUEVA: Configuración de resilencia
  static const int maxRetries = 3;
  static const Duration retryDelay = Duration(seconds: 2);
  static const int maxBufferHistory = 100; // Máximo de señales almacenadas

  final String userId;
  final String sessionId = const Uuid().v4();
  final List<UserSignal> buffer = [];
  DateTime? lastFlush;
  DateTime sessionStart = DateTime.now();
  
  // ⭐ NUEVA: Tracking de intentos fallidos
  int failedAttempts = 0;
  bool isDisposed = false;
  Timer? flushTimer;

  SignalCollector({required this.userId}) {
    lastFlush = DateTime.now();
    debugPrint('✅ SignalCollector iniciado: userId=$userId, sessionId=$sessionId');
    
    // ⭐ NUEVA: Iniciar timer de flush automático (desacoplado)
    _startAutoFlushTimer();
  }

  // ⭐ NUEVA: Timer de flush automático en background
  void _startAutoFlushTimer() {
    flushTimer = Timer.periodic(flushInterval, (_) {
      if (!isDisposed && buffer.isNotEmpty) {
        debugPrint('⏰ Auto-flush triggered by timer');
        flushToBackend();
      }
    });
  }

  // ==================== MÉTODOS ESPECÍFICOS (YA EXISTÍAN) ====================

  void trackProblemStart(String problemExpression, String topic) {
    addSignal(UserSignal(
      type: 'problem_start',
      timestamp: DateTime.now(),
      metadata: {
        'expression': problemExpression,
        'topic': topic,
      },
    ));
  }

  void trackProblemComplete(
    String problemExpression,
    int errorCount,
    Duration timeSpent,
  ) {
    addSignal(UserSignal(
      type: 'problem_complete',
      timestamp: DateTime.now(),
      metadata: {
        'expression': problemExpression,
        'error_count': errorCount,
        'time_spent_seconds': timeSpent.inSeconds,
      },
    ));
  }

  void trackInputMethod(String method) {
    addSignal(UserSignal(
      type: 'input_method',
      timestamp: DateTime.now(),
      metadata: {'method': method},
    ));
  }

  void trackError(String errorType, String context) {
    addSignal(UserSignal(
      type: 'error',
      timestamp: DateTime.now(),
      metadata: {
        'error_type': errorType,
        'context': context,
      },
    ));
  }

  void trackHintRequest() {
    addSignal(UserSignal(
      type: 'hint_request',
      timestamp: DateTime.now(),
      metadata: {},
    ));
  }

  // ⭐ NUEVO: MÉTODO GENÉRICO (MÁS FLEXIBLE)
  void trackEvent(String type, Map<String, dynamic> metadata) {
    addSignal(UserSignal(
      type: type,
      timestamp: DateTime.now(),
      metadata: metadata,
    ));
  }

  // ==================== GESTIÓN DE BUFFER ====================

  void addSignal(UserSignal signal) {
    try {
      if (isDisposed) {
        debugPrint('⚠️ SignalCollector dispuesto, ignorando señal: ${signal.type}');
        return;
      }

      buffer.add(signal);
      debugPrint('📊 Signal capturada: ${signal.type} (total buffer: ${buffer.length}/$maxBufferHistory)');

      // ⭐ NUEVA: Validar límite de buffer
      if (buffer.length > maxBufferHistory) {
        debugPrint('⚠️ Buffer excedió límite, removiendo señales antiguas');
        buffer.removeRange(0, buffer.length - maxBufferHistory);
      }

      // Flush automático si buffer lleno
      if (buffer.length >= bufferMaxSize) {
        debugPrint('⏰ Buffer lleno: ${buffer.length} >= $bufferMaxSize');
        flushToBackend();
      }
    } catch (e) {
      debugPrint('❌ Error en addSignal: $e');
    }
  }

  // ⭐ NUEVA: Retry con backoff exponencial
  Future<void> flushToBackend({int attempt = 1}) async {
    try {
      if (isDisposed) {
        debugPrint('⚠️ SignalCollector dispuesto, abortando flush');
        return;
      }

      if (buffer.isEmpty) {
        debugPrint('⚠️ Skip flush: buffer vacío');
        failedAttempts = 0; // Reset en éxito
        return;
      }

      final signalsToSend = List<UserSignal>.from(buffer);
      
      final sessionSignals = SessionSignals(
        sessionId: sessionId,
        userId: userId,
        sessionStart: sessionStart,
        signals: signalsToSend,
      );

      debugPrint('📤 [Intento $attempt/$maxRetries] Enviando ${signalsToSend.length} señales a $baseUrl/api/signals/ingest...');

      try {
        final response = await http
            .post(
              Uri.parse('$baseUrl/api/signals/ingest'),
              headers: {'Content-Type': 'application/json'},
              body: jsonEncode(sessionSignals.toJson()),
            )
            .timeout(const Duration(seconds: 5));

        if (response.statusCode == 200) {
          debugPrint('✅ Señales enviadas correctamente en intento $attempt');
          buffer.clear();
          lastFlush = DateTime.now();
          failedAttempts = 0; // ⭐ NUEVO: Reset contador en éxito
          return; // ✅ Éxito - NO reintentar
        } else if (response.statusCode == 404) {
          // ⭐ NUEVA: Error 404 = endpoint no existe, no reintentar
          debugPrint('❌ Error 404: Endpoint /api/signals/ingest no existe. Descartando buffer.');
          buffer.clear();
          failedAttempts = maxRetries; // Marcar como agotado
          return;
        } else {
          throw Exception('HTTP ${response.statusCode}');
        }
      } on TimeoutException catch (_) {
        debugPrint('⏱️ Timeout en intento $attempt');
        throw Exception('timeout');
      }
    } catch (e) {
      debugPrint('⚠️ Error en intento $attempt: $e');
      failedAttempts++;

      // ⭐ NUEVA: Reintentar si no excedió máximo
      if (attempt < maxRetries) {
        final backoffDelay = Duration(
          seconds: retryDelay.inSeconds * (1 << (attempt - 1)), // Exponencial: 2s, 4s, 8s
        );
        debugPrint('⏳ Reintentando en ${backoffDelay.inSeconds}s...');
        
        await Future.delayed(backoffDelay);
        await flushToBackend(attempt: attempt + 1);
      } else {
        debugPrint('❌ Agotados $maxRetries intentos. Buffer (${buffer.length} señales) se mantiene en memoria.');
        // ⭐ NO limpiar buffer = datos persisten para reintentar después
      }
    }
  }

  // Llamar al cerrar sesión (FUERZA envío inmediato con reintentos)
  Future<void> endSession() async {
    debugPrint('👋 Finalizando sesión y enviando señales pendientes...');
    await flushToBackend();
  }

  // Flush manual (para debug)
  Future<void> forceFlush() async {
    debugPrint('🔧 Flush forzado manualmente');
    await flushToBackend();
  }

  // ⭐ NUEVA: Cleanup seguro
  void dispose() {
    try {
      isDisposed = true;
      flushTimer?.cancel();
      debugPrint('✅ SignalCollector dispuesto. Buffer final: ${buffer.length} señales.');
    } catch (e) {
      debugPrint('⚠️ Error en dispose: $e');
    }
  }
}

}

import 'dart:convert';
import 'package:http/http.dart' as http;
import 'package:uuid/uuid.dart';

class SignalService {
  // URL del backend (cambia según tu setup)
  static const String baseUrl = 'http://192.168.0.11:8000/api/signals';
  
  // Session ID (genera uno al inicio de la app)
  static String? _sessionId;
  static String get sessionId {
    _sessionId ??= const Uuid().v4();
    return _sessionId!;
  }
  
  // User ID (hardcodeado por ahora, después usa auth real)
  static const String userId = 'user_demo_001';
  
  /// Envía una señal individual al backend
  static Future<bool> sendSignal({
    required String signalType,
    Map<String, dynamic>? metadata,
  }) async {
    try {
      final response = await http.post(
        Uri.parse('$baseUrl/ingest'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'session_id': sessionId,
          'user_id': userId,
          'signal_type': signalType,
          'metadata': metadata ?? {},
        }),
      );
      
      return response.statusCode == 200;
    } catch (e) {
      print('❌ Error enviando señal: $e');
      return false;
    }
  }
  
  /// Captura señal de problema completado
  static Future<void> problemComplete({
    required String expression,
    required int timeSpent,
    required bool correct,
  }) async {
    await sendSignal(
      signalType: 'problem_complete',
      metadata: {
        'expression': expression,
        'time_spent_seconds': timeSpent,
        'correct': correct,
      },
    );
  }
  
  /// Captura señal de error
  static Future<void> problemError({
    required String expression,
    required String errorType,
  }) async {
    await sendSignal(
      signalType: 'error',
      metadata: {
        'expression': expression,
        'error_type': errorType,
      },
    );
  }
  
  /// Captura señal de solicitud de pista
  static Future<void> hintRequest({
    required String expression,
    required int step,
  }) async {
    await sendSignal(
      signalType: 'hint_request',
      metadata: {
        'expression': expression,
        'step': step,
      },
    );
  }
  
  /// Captura método de input usado
  static Future<void> inputMethod(String method) async {
    await sendSignal(
      signalType: 'input_method',
      metadata: {'method': method}, // 'voice', 'text', 'camera'
    );
  }
}

// lib/services/visual_style_provider.dart

import 'package:flutter/foundation.dart';
import 'package:shared_preferences/shared_preferences.dart';
import '../models/visual_style.dart';

/// Provider que gestiona el estilo visual actual de la app
/// y lo persiste para futuras sesiones
class VisualStyleProvider with ChangeNotifier {
  // Estado actual
  VisualStyle _currentStyle = VisualStyle.balanced; // Default
  bool _autoAdapt = true; // Si ajusta automáticamente según comportamiento
  bool _hasCompletedOnboarding = false;

  // Keys para SharedPreferences
  static const String _keyStyle = 'visual_style';
  static const String _keyAutoAdapt = 'auto_adapt';
  static const String _keyOnboarding = 'has_completed_onboarding';

  VisualStyleProvider() {
    _loadFromPreferences();
  }

  // ==================== GETTERS ====================

  /// Estilo actual seleccionado
  VisualStyle get currentStyle => _currentStyle;

  /// Configuración completa del estilo actual
  VisualStyleConfig get config {
    switch (_currentStyle) {
      case VisualStyle.playful:
        return VisualStyleConfig.playful();
      case VisualStyle.balanced:
        return VisualStyleConfig.balanced();
      case VisualStyle.minimal:
        return VisualStyleConfig.minimal();
    }
  }

  /// Si el sistema puede adaptar automáticamente
  bool get autoAdapt => _autoAdapt;

  /// Si completó el onboarding de selección de estilo
  bool get hasCompletedOnboarding => _hasCompletedOnboarding;

  // ==================== SETTERS ====================

  /// Cambia el estilo visual
  /// 
  /// [style] - Nuevo estilo a aplicar
  /// [manual] - Si true, desactiva auto-adapt (el usuario lo eligió manualmente)
  /// [saveToPrefs] - Si guarda en SharedPreferences (default: true)
  Future<void> setStyle(
    VisualStyle style, {
    bool manual = false,
    bool saveToPrefs = true,
  }) async {
    if (_currentStyle == style) return;

    debugPrint('🎨 Cambiando estilo: $_currentStyle → $style (manual: $manual)');

    _currentStyle = style;

    // Si el usuario lo eligió manualmente, desactivar auto-adapt
    if (manual) {
      _autoAdapt = false;
      debugPrint('🔒 Auto-adapt desactivado (elección manual)');
    }

    notifyListeners();

    if (saveToPrefs) {
      await _saveToPreferences();
    }
  }

  /// Activa/desactiva la adaptación automática
  Future<void> setAutoAdapt(bool enabled) async {
    if (_autoAdapt == enabled) return;

    debugPrint('🤖 Auto-adapt: $enabled');

    _autoAdapt = enabled;
    notifyListeners();

    await _saveToPreferences();
  }

  /// Marca que completó el onboarding
  Future<void> completeOnboarding(VisualStyle chosenStyle) async {
    debugPrint('✅ Onboarding completado con estilo: $chosenStyle');

    _currentStyle = chosenStyle;
    _hasCompletedOnboarding = true;
    _autoAdapt = true; // Por default, permitir adaptación

    notifyListeners();

    await _saveToPreferences();
  }

  /// Resetea a valores por default (útil para testing)
  Future<void> reset() async {
    debugPrint('🔄 Reseteando VisualStyleProvider');

    _currentStyle = VisualStyle.balanced;
    _autoAdapt = true;
    _hasCompletedOnboarding = false;

    notifyListeners();

    await _saveToPreferences();
  }

  // ==================== PERSISTENCIA ====================

  /// Carga preferencias guardadas
  Future<void> _loadFromPreferences() async {
    try {
      final prefs = await SharedPreferences.getInstance();

      // Cargar estilo
      final styleString = prefs.getString(_keyStyle);
      if (styleString != null) {
        _currentStyle = VisualStyle.values.firstWhere(
          (e) => e.name == styleString,
          orElse: () => VisualStyle.balanced,
        );
      }

      // Cargar auto-adapt
      _autoAdapt = prefs.getBool(_keyAutoAdapt) ?? true;

      // Cargar onboarding
      _hasCompletedOnboarding = prefs.getBool(_keyOnboarding) ?? false;

      debugPrint('📂 Cargado: estilo=$_currentStyle, autoAdapt=$_autoAdapt, onboarding=$_hasCompletedOnboarding');

      notifyListeners();
    } catch (e) {
      debugPrint('❌ Error cargando preferencias: $e');
    }
  }

  /// Guarda preferencias
  Future<void> _saveToPreferences() async {
    try {
      final prefs = await SharedPreferences.getInstance();

      await prefs.setString(_keyStyle, _currentStyle.name);
      await prefs.setBool(_keyAutoAdapt, _autoAdapt);
      await prefs.setBool(_keyOnboarding, _hasCompletedOnboarding);

      debugPrint('💾 Guardado: estilo=$_currentStyle, autoAdapt=$_autoAdapt');
    } catch (e) {
      debugPrint('❌ Error guardando preferencias: $e');
    }
  }

  // ==================== HELPERS ====================

  /// Obtiene el nombre legible del estilo actual
  String get styleDisplayName {
    switch (_currentStyle) {
      case VisualStyle.playful:
        return 'Divertido';
      case VisualStyle.balanced:
        return 'Balanceado';
      case VisualStyle.minimal:
        return 'Minimalista';
    }
  }

  /// Obtiene descripción del estilo actual
  String get styleDescription {
    switch (_currentStyle) {
      case VisualStyle.playful:
        return 'Colores brillantes y animaciones alegres';
      case VisualStyle.balanced:
        return 'Equilibrio entre visual y funcionalidad';
      case VisualStyle.minimal:
        return 'Diseño sobrio y directo';
    }
  }

  /// Obtiene el ícono del estilo actual
  String get styleIcon {
    switch (_currentStyle) {
      case VisualStyle.playful:
        return '🎨';
      case VisualStyle.balanced:
        return '⚖️';
      case VisualStyle.minimal:
        return '▫️';
    }
  }
}
