    def generate_explanation(

- Usá SOLO estrategias que le funcionaron antes
- Evitá completamente las que fallaron
- Ajustá tu tono al estilo visual: {user_context.get('visual_style', 'balanced')}
- Hablá en argentino (vos, querés, probá, etc.)
"""
        
        user_prompt = f"""Problema actual: {problem.get('expression', '')}
Tipo de intervención necesaria: {intervention_type}

Generá en formato JSON:
{{
  "speech": "texto que Banny va a decir por voz",
  "visual_cue": {{"type": "highlight/step/animation", "timing": "immediate/after_pause"}},
  "followup": "pregunta opcional si hace falta"
}}
"""

        try:
            response = self.client.chat.completions.create(
                model="gpt-4o-mini",  # Más barato que gpt-4
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_prompt}
                ],
                temperature=0.7,
                max_tokens=300
            )
            
            result = json.loads(response.choices[0].message.content)
            return result
            
        except json.JSONDecodeError:
            # Si la respuesta no es JSON válido
            return {
                "speech": response.choices[0].message.content[:200],
                "visual_cue": {"type": "step", "timing": "immediate"},
                "followup": ""
            }
        except Exception as e:
            print(f"❌ Error OpenAI: {e}")
            return self._generate_fallback_response(problem)
    
    def _generate_fallback_response(self, problem: Dict) -> Dict:
        """
        Respuesta genérica si OpenAI falla
        """
        return {
            "speech": f"Veamos el problema {problem.get('expression', '')}. ¿Qué querés hacer primero?",
            "visual_cue": {"type": "highlight", "timing": "immediate"},
            "followup": "¿Por dónde empezarías?"
        }
    
    def _build_memory_context(self, similar_successes: Dict) -> Dict:
        """
        Construye contexto de memoria pedagógica
        """
        if not similar_successes['metadatas'] or not similar_successes['metadatas'][0]:
            return {
                "successful_approaches": "Ninguno aún (primera vez con este tipo de problema)",
                "failed_approaches": "Ninguno"
            }
        
        # Filtrar por effectiveness > 0.7
        successes = [
            m.get('strategy', '') 
            for m in similar_successes['metadatas'][0]
            if m.get('effectiveness', 0) > 0.7
        ]
        
        # Filtrar failures (effectiveness == 0.0)
        failures = [
            m.get('strategy', '') 
            for m in similar_successes['metadatas'][0]
            if m.get('effectiveness', 0) == 0.0
        ]
        
        return {
            "successful_approaches": ", ".join(successes) if successes else "Ninguno todavía",
            "failed_approaches": ", ".join(failures) if failures else "Ninguno"
        }

# backend/intelligence/user_profile.py (NUEVO ARCHIVO)

import chromadb
from sentence_transformers import SentenceTransformer
from typing import List, Dict, Optional
from datetime import datetime
import json

class UserIntelligenceProfile:
    """
    Memoria vectorial personalizada con ChromaDB
    """
    
    def __init__(self):
        # ChromaDB client (persistente en disco)
        self.chroma_client = chromadb.PersistentClient(path="./chroma_db")
        
        # Colección para guardar interacciones exitosas
        self.collection = self.chroma_client.get_or_create_collection(
            name="user_profiles",
            metadata={"description": "Pedagogical memory for MathTutor"}
        )
        
        # Modelo de embeddings (ligero, 80MB)
        self.embedder = SentenceTransformer('all-MiniLM-L6-v2')
    
    def store_successful_interaction(
        self,
        user_id: str,
        problem: Dict,
        solution_strategy: str,
        explanation_used: str,
        effectiveness_score: float,  # 0.0-1.0
        visual_style_used: str,
        scaffolding_level: float,
    ):
        """
        Guarda cada ÉXITO para reutilizarlo después
        """
        # Crear embedding del problema
        problem_text = f"{problem.get('type', '')} {problem.get('expression', '')} {problem.get('topic', '')}"
        embedding = self.embedder.encode(problem_text).tolist()
        
        # ID único para esta interacción
        interaction_id = f"{user_id}_{int(datetime.now().timestamp())}"
        
        # Almacenar en ChromaDB
        self.collection.add(
            embeddings=[embedding],
            documents=[explanation_used],
            metadatas=[{
                "user_id": user_id,
                "problem_type": problem.get('type', 'unknown'),
                "strategy": solution_strategy,
                "effectiveness": effectiveness_score,
                "timestamp": datetime.now().isoformat(),
                "visual_style_used": visual_style_used,
                "scaffolding_level": scaffolding_level,
            }],
            ids=[interaction_id]
        )
        
        print(f"✅ Stored success for {user_id}: {problem_text[:50]}...")
    
    def retrieve_similar_successes(
        self,
        user_id: str,
        current_problem: Dict,
        top_k: int = 3
    ) -> List[Dict]:
        """
        Busca éxitos pasados similares al problema actual
        """
        # Crear embedding del problema actual
        problem_text = f"{current_problem.get('type', '')} {current_problem.get('expression', '')} {current_problem.get('topic', '')}"
        query_embedding = self.embedder.encode(problem_text).tolist()
        
        # Búsqueda vectorial en ChromaDB
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=top_k,
            where={"user_id": user_id}  # Solo del mismo usuario
        )
        
        # Si NO tiene historial propio, buscar de usuarios similares
        if not results['documents'][0]:
            print(f"⚠️ No history for {user_id}, searching similar users...")
            results = self.collection.query(
                query_embeddings=[query_embedding],
                n_results=top_k,
                # Sin filtro de usuario
            )
        
        return results
    
    def store_failed_interaction(
        self,
        user_id: str,
        problem: Dict,
        failed_approach: str,
        reason: str
    ):
        """
        Guarda qué NO funcionó para evitarlo en el futuro
        """
        # Guardar en metadata como "effectiveness": 0.0
        problem_text = f"{problem.get('type', '')} {problem.get('expression', '')}"
        embedding = self.embedder.encode(problem_text).tolist()
        
        interaction_id = f"{user_id}_FAIL_{int(datetime.now().timestamp())}"
        
        self.collection.add(
            embeddings=[embedding],
            documents=[f"FAILED: {failed_approach}"],
            metadatas=[{
                "user_id": user_id,
                "problem_type": problem.get('type', 'unknown'),
                "strategy": failed_approach,
                "effectiveness": 0.0,  # MARCA DE FALLO
                "failure_reason": reason,
                "timestamp": datetime.now().isoformat(),
            }],
            ids=[interaction_id]
        )
        
        print(f"❌ Stored failure for {user_id}: {failed_approach}")

# backend/prediction/intervention_engine.py (NUEVO ARCHIVO)

from typing import Dict, Optional, List
import numpy as np
from datetime import datetime

class InterventionEngine:
    """
    Predice confusión ANTES del error y decide QUÉ hacer
    """
    
    def __init__(self, user_profile):
        self.profile = user_profile
        self.threshold_confusion = 0.65  # Ajustable por usuario
    
    def predict_struggle_probability(
        self,
        user_id: str,
        current_problem: Dict,
        current_time_spent: float,  # segundos
        session_signals: List[Dict]  # últimas 10 señales
    ) -> float:
        """
        Predice probabilidad de confusión (0-1)
        """
        # 1. Buscar problemas similares en historial
        similar_past = self.profile.retrieve_similar_successes(
            user_id, 
            current_problem,
            top_k=5
        )
        
        # 2. Calcular tiempo esperado basado en historial
        if similar_past['metadatas'] and similar_past['metadatas'][0]:
            # Extraer tiempos de problemas similares (simulado, ajustar según tu metadata)
            historical_times = [15, 20, 18]  # Placeholder: deberías guardarlo en metadata
            expected_time = np.mean(historical_times)
            time_ratio = current_time_spent / expected_time
        else:
            time_ratio = 1.0  # Sin historial, usar heursticas generales
        
        # 3. Analizar señales recientes de frustración
        frustration_signals = self._detect_frustration_signals(session_signals[-10:])
        
        # 4. Calcular probabilidad ponderada
        struggle_prob = (
            0.4 * min(time_ratio, 2.0) +  # 40% peso: tiempo vs. esperado
            0.3 * frustration_signals +    # 30% peso: señales emocionales
            0.3 * (1 - self._calc_historical_success_rate(similar_past))
        )
        
        return min(struggle_prob, 1.0)
    
    def _detect_frustration_signals(self, signals: List[Dict]) -> float:
        """
        Detecta frustración por: taps rápidos, errores repetidos, pausas largas
        """
        if not signals:
            return 0.0
        
        frustration_score = 0.0
        
        # Contar errores consecutivos
        errors = [s for s in signals if s.get('type') == 'error']
        if len(errors) >= 3:
            frustration_score += 0.5
        
        # Detectar taps rápidos (< 1s entre taps)
        # ... (implementar lógica)
        
        return min(frustration_score, 1.0)
    
    def _calc_historical_success_rate(self, similar_past: Dict) -> float:
        """
        Calcula tasa de éxito histórica en problemas similares
        """
        if not similar_past['metadatas'] or not similar_past['metadatas'][0]:
            return 0.5  # Sin historial, asumir 50%
        
        # Calcular promedio de effectiveness
        effectiveness_scores = [
            m.get('effectiveness', 0.5) 
            for m in similar_past['metadatas'][0]
        ]
        
        return np.mean(effectiveness_scores) if effectiveness_scores else 0.5
    
    def decide_intervention(
        self,
        struggle_probability: float,
        user_preferences: Dict,
        current_context: Dict
    ) -> Optional[Dict]:
        """
        Decide QUÉ hacer y CUÁNDO
        """
        if struggle_probability < 0.3:
            return None  # Usuario va bien, no intervenir
        
        elif struggle_probability < 0.6:
            # Intervención SUAVE: visual hint sin interrumpir
            return {
                "type": "subtle_visual_cue",
                "action": "highlight_relevant_part",
                "timing": "immediate",
                "intensity": 0.4,
            }
        
        elif struggle_probability < 0.8:
            # Intervención MEDIA: pregunta socrática
            similar_successes = self.profile.retrieve_similar_successes(
                user_preferences['user_id'],
                current_context['problem'],
                top_k=1
            )
            
            # Usar estrategia que funcionó antes
            if similar_successes['metadatas'] and similar_successes['metadatas'][0]:
                best_approach = similar_successes['metadatas'][0][0].get('strategy', 'generic')
                question = self._generate_socratic_question(best_approach)
            else:
                question = self._generate_generic_clarification()
            
            return {
                "type": "socratic_question",
                "action": "change_to_portrait",
                "content": question,
                "timing": "after_next_pause",
                "intensity": 0.65,
            }
        
        else:  # >= 0.8
            # Intervención FUERTE: ejemplo guiado
            return {
                "type": "guided_example",
                "action": "show_similar_solved_problem",
                "scaffolding_level": 0.9,
                "timing": "immediate",
                "intensity": 0.95,
            }
    
    def _generate_socratic_question(self, approach: str) -> str:
        """
        Genera pregunta socrática basada en estrategia exitosa previa
        """
        questions = {
            "algebraic": "¿Qué operación necesitás hacer para dejar x sola?",
            "visual": "Si movés ese número, ¿a dónde va?",
            "trial_error": "Probá con un número más simple primero, ¿qué pasa con 2?",
        }
        return questions.get(approach, "¿Qué estás tratando de hacer con ese término?")
    
    def _generate_generic_clarification(self) -> str:
        return "¿Por qué elegiste hacer eso? Explicame tu razonamiento."
