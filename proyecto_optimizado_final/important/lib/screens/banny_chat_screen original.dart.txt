  bool isTyping = false;

  late Animation<double> cardSlideAnimation;

  // ✅ Servicios de hiperpersonalización (CORREGIDO)

  late SignalCollector signalCollector; // ✅ CAMBIO: SignalCollector en vez de SignalCollectorService

  @override
  void initState() {
    super.initState();
    addInitialMessage();

      duration: const Duration(milliseconds: 600),
      vsync: this,

      CurvedAnimation(parent: cardAnimationController, curve: Curves.elasticOut),
    );

    cardScaleAnimation = Tween<double>(begin: 0.8, end: 1.0).animate(
      CurvedAnimation(parent: cardAnimationController, curve: Curves.easeOutBack),
    );
    
    // ⭐ CAMBIO: Inicializar signal tracking DE FORMA COMPLETAMENTE INDEPENDIENTE
    _initializeSignalTracking();
    
    // ⭐ CAMBIO: Inicializar WebSocket DE FORMA COMPLETAMENTE INDEPENDIENTE
    _initializeWebSocket();
  }
  
  /// ⭐ NUEVA: Inicializar signal tracking (sin bloquear, sin depender de WebSocket)
  void _initializeSignalTracking() {
    try {
      signalCollector = SignalCollector(userId: widget.userId);
      
      // Trackear inicio de sesión
      signalCollector.trackEvent('session_start', {
        'timestamp': DateTime.now().toIso8601String(),
        'user_id': widget.userId,
      });
      
      print('✅ Signal tracking inicializado para ${widget.userId}');
    } catch (e) {
      print('❌ Error en _initializeSignalTracking: $e');
    }
  }
  
  /// ⭐ NUEVA: Inicializar WebSocket (completamente desacoplado de signals)
  void _initializeWebSocket() {
    // Ejecutar en background, sin bloquear UI
    Future.microtask(() async {
      try {
        print('🔌 Iniciando WebSocket en background...');
        
        tutorService = HyperpersonalizedTutorService(
          userId: widget.userId,
          onTutorAction: _handleTutorAction,
        );
        
        // Esperar conexión con timeout
        await tutorService!.connect().timeout(
          const Duration(seconds: 8),
          onTimeout: () {
            print('⏱️ WebSocket timeout');
            if (mounted) {
              setState(() => isOnline = false);
            }
          },
        );
        
        // Si llegamos aquí, la conexión fue exitosa
        if (mounted) {
          setState(() => isOnline = true);
        }
        
        // Escuchar cambios de conexión
        tutorService?.connectionStream.listen(
          (isConnected) {
            if (mounted) {
              setState(() => isOnline = isConnected);
            }
          },
          onError: (e) {
            print('⚠️ connectionStream error: $e');
            if (mounted) {
              setState(() => isOnline = false);
            }
          },
        );
        
      } catch (e) {
        print('❌ Error en _initializeWebSocket: $e');
        if (mounted) {
          setState(() => isOnline = false);
        }
      }
    });
  }
  
  /// Manejar TutorAction del backend
  void _handleTutorAction(TutorAction action) {
    print('🎯 Ejecutando TutorAction: ${action.type}');
    
    setState(() {
      isTyping = true;
    });
    
    Future.delayed(const Duration(milliseconds: 500), () {
      if (!mounted) return;
      
      setState(() {
        isTyping = false;
        
        messages.add(BannyChatMessage(
          id: DateTime.now().toString(),
          text: action.speech,
          isUser: false,
          timestamp: DateTime.now(),
        ));
        
        if (action.followup != null && action.followup!.isNotEmpty) {
          Future.delayed(const Duration(milliseconds: 800), () {
            if (!mounted) return;
            setState(() {
              messages.add(BannyChatMessage(
                id: '${DateTime.now().toString()}_followup',
                text: action.followup!,
                isUser: false,
                timestamp: DateTime.now(),
              ));
            });
            scrollToBottom();
          });
        }
      });
      
      scrollToBottom();
      
      if (action.visualCue != null) {
        _executeVisualCue(action.visualCue!);
      }
      
      signalCollector.trackEvent('intervention_received', {
        'type': action.type,
        'intensity': action.intensity,
        'struggle_probability': action.struggleprobability,
      });
    });
  }

  }

  void addInitialMessage() {
    setState(() {

  Future<void> solveProblem(String expression) async {
    final mathPattern = RegExp(r'[-+*/=xX0-9() ]+');

    if (!mathPattern.hasMatch(expression)) {
      addMessage('Por favor, escribe una expresión matemática válida 🤔', false);
      return;
    }

    setState(() {
      currentExpression = expression;
      isLoading = true;
      isTyping = true;
      showBoard = true;
    });
    
    // ✅ Trackear inicio de problema
    try {
      signalCollector.trackEvent('problem_start', {
        'expression': expression,
        'timestamp': DateTime.now().toIso8601String(),
      });
    } catch (e) {
      print('⚠️ Error trackando inicio de problema: $e');
    }

    setState(() {
      messages.add(BannyChatMessage(
        id: DateTime.now().toString(),
        text: '¡Vamos a resolverlo juntos!',
        isUser: false,
        timestamp: DateTime.now(),
        mathExpression: expression,
        isProblemCard: true,
      ));
    });

    scrollToBottom();
    cardAnimationController.forward(from: 0);

    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (mounted && sheetController.isAttached) {
        sheetController.animateTo(
          0.35,
          duration: const Duration(milliseconds: 500),
          curve: Curves.easeOut,
        );
      }
    });

    try {
      final response = await MathApiService.solveMath(
        expression: expression,
        showSteps: true,
      );

      final solution = MathSolution.fromJson(response);

      if (!mounted) return;

      setState(() {
        currentSolution = solution;
        isLoading = false;
        isTyping = false;
      });
      
      // ✅ Trackear problema completado
      try {
        signalCollector.trackEvent('problem_complete', {
          'expression': expression,
          'solution': solution.solution,
          'steps_count': solution.steps.length,
          'success': true,
        });
      } catch (e) {
        print('⚠️ Error trackando completitud: $e');
      }

      await Future.delayed(const Duration(milliseconds: 300));

      if (mounted && sheetController.isAttached) {
        sheetController.animateTo(
          0.75,
          duration: const Duration(milliseconds: 400),
          curve: Curves.easeOut,
        );
      }
      
    } catch (e) {
      if (!mounted) return;

      setState(() {
        isLoading = false;
        isTyping = false;
      });

      addMessage('Error al resolver. ¿Está el backend corriendo? 🔧', false);
      
      // ✅ Trackear error
      try {
        signalCollector.trackEvent('problem_error', {
          'expression': expression,
          'error': e.toString(),
        });
      } catch (trackError) {
        print('⚠️ Error trackando error: $trackError');
      }

      debugPrint('Error backend: $e');
    }
  }

  void addMessage(String text, bool isUser) {
    setState(() {
      messages.add(BannyChatMessage(
        id: DateTime.now().toString(),
        text: text,
        isUser: isUser,
        timestamp: DateTime.now(),
      ));
    });
    scrollToBottom();
  }

  void sendMessage() {
    try {
      final text = messageController.text.trim();

      if (text.isEmpty) return;

      addMessage(text, true);
      messageController.clear();
      
      // ✅ Trackear mensaje enviado (con manejo de errores)
      try {
        signalCollector.trackEvent('message_sent', {
          'text': text,
          'length': text.length,
        });
      } catch (e) {
        print('⚠️ Error trackando mensaje: $e');
      }

      solveProblem(text);
    } catch (e) {
      print('❌ Error en sendMessage: $e');
      debugPrint('Stack: $e');
    }
  }

  void scrollToBottom() {
    Future.delayed(const Duration(milliseconds: 100), () {
      if (chatScrollController.hasClients) {
        chatScrollController.animateTo(
          chatScrollController.position.maxScrollExtent,
          duration: const Duration(milliseconds: 300),
          curve: Curves.easeOut,
        );
      }
    });
  }

  @override
  void dispose() {
    try {
      messageController.dispose();
      chatScrollController.dispose();
      cardAnimationController.dispose();
      
      // ⭐ CAMBIO: Signal tracking completamente independiente
      signalCollector.trackEvent('session_end', {
        'timestamp': DateTime.now().toIso8601String(),
      });
      
      // ⭐ CAMBIO: Llamar dispose de signalCollector
      signalCollector.dispose();
      
      // Enviar señales finales (flush final)
      signalCollector.flushToBackend().then((_) {
        print('✅ Señales finales enviadas');
      }).catchError((e) {
        print('⚠️ Error enviando señales finales: $e');
      });
      
      // Solo disponer tutorService si fue inicializado
      if (tutorService != null) {
        try {
          tutorService!.dispose();
        } catch (e) {
          print('⚠️ Error en tutorService.dispose(): $e');
        }
      }
    } catch (e) {
      print('⚠️ Error en dispose: $e');
    }
    
    super.dispose();
  }

  Widget buildChatMessages() {
    return ListView.builder(
      controller: chatScrollController,
      padding: EdgeInsets.only(
        left: 16,
        right: 16,
        top: 12,
        bottom: showBoard ? 450 : 100,
      ),
      itemCount: messages.length + (isTyping && !isLoading ? 1 : 0),
      itemBuilder: (context, index) {
        if (isTyping && !isLoading && index == messages.length) {
          return buildTypingIndicator();
        }
        return buildMessageBubble(messages[index]);
      },
    );
  }
